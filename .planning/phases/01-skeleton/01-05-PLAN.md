- GET /api/setup/state → {initialized: boolean, version: string}
- POST /api/setup/initialize {email, password, display_name, workspace_name} → 200 + Set-Cookie

- POST /api/auth/login {email, password} → 200 + Set-Cookie
- POST /api/auth/logout → 200
- GET /api/auth/me → {user, workspaces, current_workspace, role_in_current_workspace}
- POST /api/me/workspaces/{ws_id}/switch → 200 + new Set-Cookie
- GET /api/invites/list → admin 限定，[{id, email, role_code, status, expires_at, ...}]
- POST /api/invites {email, role_code} → admin 限定，200 (msg)
- GET /api/invites/accept?token= → 200 {workspace_name, role_code, inviter_display_name, expires_at, already_used}
- POST /api/invites/accept {token, password, display_name, department?} → 200 + Set-Cookie

       ```typescript
       const API_BASE = process.env.NEXT_PUBLIC_API_BASE ?? '/api';
       export interface ApiCallOptions { method?: string; body?: unknown; signal?: AbortSignal; }
       export async function apiCall<T>(path: string, opts: ApiCallOptions = {}): Promise<T> {
         const res = await fetch(`${API_BASE}${path}`, {
           method: opts.method ?? 'GET',
           credentials: 'include',
           headers: opts.body ? { 'Content-Type': 'application/json' } : {},
           body: opts.body ? JSON.stringify(opts.body) : undefined,
           signal: opts.signal,
         });
         if (!res.ok) {
           const err: ApiError = await res.json().catch(() => ({ error: res.statusText }));
           if (res.status === 401 && !path.startsWith('/auth/login') && !path.startsWith('/setup/')) {
             window.location.href = '/login';
           }
           throw new ApiError(err.error || res.statusText, res.status, err);
         }
         return res.json();
       }
       export class ApiError extends Error { constructor(public message: string, public status: number, public detail: any) { super(message); } }
       ```

       ```typescript
         const store = useUserStore();
         const query = useQuery({ queryKey: ['me'], queryFn: () => apiCall<MeResponse>('/auth/me'), enabled: !store.user });
         useEffect(() => { if (query.data) store.setMe(query.data); }, [query.data]);
         return { user: store.user, workspaces: store.workspaces, role: store.role, currentWorkspace: store.currentWorkspace, loading: query.isLoading };
       }
       ```

       ```typescript
       import { NextRequest, NextResponse } from 'next/server';
       export async function middleware(req: NextRequest) {
         const path = req.nextUrl.pathname;
         if (path.startsWith('/_next/') || path.startsWith('/api/') || path === '/favicon.ico') return NextResponse.next();
         const res = await fetch(`${process.env.INTERNAL_API_BASE}/api/setup/state`, { headers: { cookie: req.headers.get('cookie') ?? '' } });
         if (!res.ok) return NextResponse.next();
         const data = await res.json();
         if (!data.initialized && path !== '/setup') return NextResponse.redirect(new URL('/setup', req.url));
         if (data.initialized && path === '/setup') return NextResponse.redirect(new URL('/', req.url));
         return NextResponse.next();
       }
       export const config = { matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'] };
       ```
