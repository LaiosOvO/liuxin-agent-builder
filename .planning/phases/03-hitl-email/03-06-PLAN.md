from __future__ import annotations
import hmac
import hashlib
import logging
from typing import Annotated

from fastapi import APIRouter, Request, Response, HTTPException, Depends, Form, status
from fastapi.responses import HTMLResponse, RedirectResponse
from fastapi.templating import Jinja2Templates
from pathlib import Path
from sqlalchemy.ext.asyncio import AsyncSession

from app.services.hitl_token_service import HitlTokenService, InvalidSignature, TokenExpired, InvalidAudience, HitlTokenError
from app.agent_builder.security.bot_detector import is_bot_ua
from app.agent_builder.security.rate_limit import limiter
from app.agent_builder.api.deps import get_db
from app.agent_builder.workflow.hitl_token_store import HitlTokenStore
from app.agent_builder.services.hitl_action_service import HitlActionService

log = logging.getLogger(__name__)

router = APIRouter(prefix="/hitl", tags=["hitl"])

_TEMPLATE_DIR = Path(__file__).parent.parent.parent.parent / "templates" / "hitl"
templates = Jinja2Templates(directory=str(_TEMPLATE_DIR))


@router.get("/page/{token}", response_class=HTMLResponse)
@limiter.limit("60/minute")  # NET-03 速率限制
async def hitl_page_view(
    request: Request,
    token: str,
    db: Annotated[AsyncSession, Depends(get_db)],
):
    """GET 决策页 — Phase 3 P0 价值演示入口。

    CLAUDE.md 2.5：GET 不消费 jti（永不可接受 GET 消费）
    """
    ua = request.headers.get("User-Agent", "")
    ip = request.client.host if request.client else ""

    # 0. Bot UA 检测 — Pitfall 3 P0 防护
    if is_bot_ua(ua):
        log.info(f"bot UA detected, returning static scan page: ua={ua[:80]}")
        return templates.TemplateResponse(
            "bot_scan.html",
            {"request": request},
            status_code=200,
        )

    # 1. JWT decode（仅校签 + exp + aud）
    try:
        payload = HitlTokenService.decode(token)
    except TokenExpired:
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "链接已过期，请联系发起人重新发起"}, status_code=410
        )
    except (InvalidSignature, InvalidAudience):
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "无效的链接签名"}, status_code=401
        )
    except HitlTokenError as e:
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": str(e)}, status_code=400
        )

    # 2. 检查 jti 已消费（仅查询，不消费）
    store = HitlTokenStore(db, request.app.state.redis)
    from uuid import UUID
    if await store.is_consumed(UUID(payload["jti"])):
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "决策已提交，链接失效"}, status_code=410
        )

    # 3. 加载节点状态（form_schema + records + deadline）
    from app.agent_builder.models.node_state import NodeState
    ns = await db.get(NodeState, UUID(payload["node_state_id"]))
    if not ns:
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "节点不存在"}, status_code=404
        )

    # 4. 签 30min session cookie（HMAC{jti}）
    cookie_value = _sign_session_cookie(payload["jti"])

    # 5. 渲染表单
    response = templates.TemplateResponse(
        "page.html",
        {
            "request": request,
            "token": token,
            "jti": payload["jti"],
            "action": payload["allowed_actions"][0],
            "node_state_id": payload["node_state_id"],
            "form_schema": ns.payload.get("form_schema", {}),
            "records": ns.payload.get("records", []),
            "deadline_at": ns.payload.get("deadline_at"),
            "current_actor": ns.payload.get("current_actor"),
            "phase": ns.payload.get("phase"),
        },
        status_code=200,
    )
    response.set_cookie(
        key=f"hitl_session_{payload['jti']}",
        value=cookie_value,
        max_age=1800,  # 30 min
        httponly=True,
        secure=True,
        samesite="lax",
    )
    return response


@router.post("/action/{token}", response_class=HTMLResponse)
@limiter.limit("10/minute")  # 防暴力提交
async def hitl_action_submit(
    request: Request,
    token: str,
    action: Annotated[str, Form()],
    reason: Annotated[str, Form()] = "",
    db: Annotated[AsyncSession, Depends(get_db)] = None,
):
    """POST 提交决策 — 消费 jti + advisory_lock + Command(resume)。"""
    ua = request.headers.get("User-Agent", "")
    ip = request.client.host if request.client else ""

    # 1. JWT decode
    try:
        payload = HitlTokenService.decode(token)
    except (TokenExpired, InvalidSignature, InvalidAudience, HitlTokenError) as e:
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "链接已失效"}, status_code=410
        )

    # 2. cookie 校验
    cookie_value = request.cookies.get(f"hitl_session_{payload['jti']}")
    if not cookie_value or not _verify_session_cookie(payload["jti"], cookie_value):
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "会话已过期，请重新点击邮件链接"}, status_code=401
        )

    # 3. form_data 从 Form 字段读 + form_schema 校验（jsonschema）
    form_data = {k: v for k, v in (await request.form()).items() if k not in ("action", "reason")}

    # 4. 业务层：advisory_lock + jti consume + resume + sibling invalidate + audit
    svc = HitlActionService(db, request.app.state.redis, request.app.state.checkpointer)
    try:
        result = await svc.submit_action(
            payload=payload,
            action=action,
            reason=reason,
            form_data=form_data,
            ip=ip,
            ua=ua,
        )
    except JtiAlreadyConsumed:
        return templates.TemplateResponse(
            "error.html", {"request": request, "error": "决策已提交，无法重复提交"}, status_code=409
        )
    except FormDataValidationError as e:
        return templates.TemplateResponse(
            "page.html", {"request": request, "errors": e.errors, ...}, status_code=422
        )

    # 5. 成功页
    return templates.TemplateResponse(
        "success.html",
        {"request": request, "action": action, "instance_id": payload["flow_id"]},
        status_code=200,
    )


# --- session cookie 工具 ---
import os
def _sign_session_cookie(jti: str) -> str:
    secret = os.environ["HMAC_SECRET"].encode()
    sig = hmac.new(secret, jti.encode(), hashlib.sha256).hexdigest()
    return f"{jti}:{sig}"

def _verify_session_cookie(jti: str, cookie_value: str) -> bool:
    expected = _sign_session_cookie(jti)
    return hmac.compare_digest(expected, cookie_value)

# backend/app/agent_builder/services/hitl_action_service.py
"""HITL 决策提交业务逻辑层 — 并发安全核心。

设计参考 reading-dify-03-06-hitl-api-2026-05-17.md §7：
- advisory_lock (thread_id) + jti 原子消费 = 双保险
- LangGraph graph.ainvoke(Command(resume=...)) 推进 interrupt
"""

from __future__ import annotations
import logging
from typing import Any
from uuid import UUID

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession
from langgraph.types import Command
from jsonschema import ValidationError as JsonSchemaError

from app.agent_builder.workflow.hitl_token_store import HitlTokenStore
from app.agent_builder.workflow.checkpoint import build_thread_id
from app.agent_builder.workflow.hitl_payload import validate_form_data, append_record, compute_next_status
from app.agent_builder.models.node_state import NodeState
from app.agent_builder.models.audit_log import AuditLog
from app.agent_builder.models.flow_instance import FlowInstance

log = logging.getLogger(__name__)


class JtiAlreadyConsumed(Exception):
    """jti 已被消费（409）。"""


class FormDataValidationError(Exception):
    def __init__(self, errors: list[str]):
        self.errors = errors


class HitlActionService:
    def __init__(self, db: AsyncSession, redis, checkpointer):
        self.db = db
        self.redis = redis
        self.checkpointer = checkpointer

    async def submit_action(
        self,
        *,
        payload: dict,
        action: str,
        reason: str,
        form_data: dict,
        ip: str,
        ua: str,
    ) -> dict[str, Any]:
        """完整提交链路：lock → consume jti → invalidate siblings → resume graph → audit。"""

        jti = UUID(payload["jti"])
        node_state_id = UUID(payload["node_state_id"])
        instance_id = UUID(payload["flow_id"])
        actor_id = UUID(payload["actor_id"])

        # 加载 instance 拿 workspace_id（thread_id 用）
        flow_instance = await self.db.get(FlowInstance, instance_id)
        if not flow_instance:
            raise ValueError("instance not found")
        thread_id = build_thread_id(flow_instance.workspace_id, instance_id)

        # 1. Advisory lock（Pitfall 2 防护）
        # 用 thread_id hash 作为 lock_key（保持单实例串行）
        lock_key = hash(thread_id) & 0x7FFFFFFFFFFFFFFF  # 64-bit signed
        await self.db.execute(text("SELECT pg_advisory_xact_lock(:k)"), {"k": lock_key})

        # 2. form_schema 校验 form_data
        ns = await self.db.get(NodeState, node_state_id)
        form_schema = ns.payload.get("form_schema", {})
        if form_schema and form_data:
            try:
                validate_form_data(form_schema, form_data)
            except JsonSchemaError as e:
                raise FormDataValidationError([e.message])

        # 3. 原子消费 jti（HitlTokenStore.consume — UPDATE...WHERE used_at IS NULL RETURNING）
        store = HitlTokenStore(self.db, self.redis)
        token_row = await store.consume(jti, ip=ip, ua=ua)
        if token_row is None:
            raise JtiAlreadyConsumed()

        # 4. 失效同节点其他 token
        invalidated = await store.invalidate_siblings(node_state_id, except_jti=jti)
        log.info(f"invalidated {invalidated} sibling tokens for node_state_id={node_state_id}")

        # 5. 更新 node_state.payload（append record + 改 status）
        new_payload = append_record(
            ns.payload,
            actor_id=actor_id,
            actor_email=payload.get("actor_email", ""),
            action=action,
            reason=reason,
            form_data=form_data,
            ip=ip,
            ua=ua,
        )
        new_status = compute_next_status(action, ns.payload["phase"])
        ns.payload = new_payload
        ns.status = new_status

        # 6. LangGraph resume
        graph = await self._load_graph_for_instance(flow_instance)
        await graph.ainvoke(
            Command(resume={
                "action": action,
                "reason": reason,
                "form_data": form_data,
                "actor_id": str(actor_id),
                "ip": ip,
                "ua": ua,
                "jti": str(jti),
            }),
            config={"configurable": {"thread_id": thread_id}},
        )

        # 7. 写 audit_log（NET-05）
        audit = AuditLog(
            workspace_id=flow_instance.workspace_id,
            actor_user_id=actor_id,
            action="hitl.decision",
            object_type="node_state",
            object_id=str(node_state_id),
            actor_ip=ip,
            actor_ua=ua,
            decision=action,
            node_state_id=node_state_id,
            metadata={"jti": str(jti), "instance_id": str(instance_id), "reason": reason},
        )
        self.db.add(audit)

        await self.db.commit()

        return {"status": "ok", "new_node_status": new_status}

    async def _load_graph_for_instance(self, instance: FlowInstance):
        from app.agent_builder.workflow.compiler import DSLCompiler
        from app.agent_builder.models.workflow_version import WorkflowVersion
        wv = await self.db.get(WorkflowVersion, instance.workflow_version_id)
        compiler = DSLCompiler(
            workspace_id=instance.workspace_id,
            instance_id=instance.id,
            redis=self.redis,
        )
        compiled = compiler.compile(wv.dsl, checkpointer=self.checkpointer)
        return compiled.graph
