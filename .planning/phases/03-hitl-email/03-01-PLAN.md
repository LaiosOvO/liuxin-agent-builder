```python
from __future__ import annotations
from datetime import datetime
from uuid import UUID, uuid4

import sqlalchemy as sa
from sqlalchemy import Index, String, DateTime, ForeignKey
from sqlalchemy.dialects.postgresql import UUID as PGUUID
from sqlalchemy.orm import Mapped, mapped_column

from app.agent_builder.models.base import Base, TimestampMixin

class HitlToken(TimestampMixin, Base):
    """HITL 决策 token（jti 一次性消费）。

    设计参考 docs/reading-dify-03-01-hitl-schema-2026-05-17.md：
    - Dify HumanInputForm 三表分离 vs 我们 hitl_tokens 单表设计权衡
    - jti = JWT payload 中 jti claim（UUID），用作 PK
    - used_at IS NULL 表示未消费；UPDATE...WHERE used_at IS NULL RETURNING * 实现幂等消费
    """

    __tablename__ = "hitl_tokens"
    __table_args__ = (
        Index("ix_hitl_tokens_node_state_used", "node_state_id", "used_at"),
        Index("ix_hitl_tokens_actor_exp", "actor_id", "expires_at"),
        Index("ix_hitl_tokens_instance_action", "instance_id", "action"),
    )

    jti: Mapped[UUID] = mapped_column(PGUUID(as_uuid=True), primary_key=True, default=uuid4)
    instance_id: Mapped[UUID] = mapped_column(PGUUID(as_uuid=True), ForeignKey("flow_instances.id", ondelete="CASCADE"), nullable=False)
    node_state_id: Mapped[UUID] = mapped_column(PGUUID(as_uuid=True), ForeignKey("node_states.id", ondelete="CASCADE"), nullable=False)
    actor_id: Mapped[UUID] = mapped_column(PGUUID(as_uuid=True), ForeignKey("users.id", ondelete="RESTRICT"), nullable=False)
    action: Mapped[str] = mapped_column(String(16), nullable=False)  # submit / approve / return / reject

    expires_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), nullable=False)
    used_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
    used_ip: Mapped[str | None] = mapped_column(String(64), nullable=True)
    used_ua: Mapped[str | None] = mapped_column(String(256), nullable=True)
```

```python
# backend/app/agent_builder/workflow/hitl_token_store.py
"""HITL jti 黑名单 — Redis 加速 + Postgres 权威。

设计：
- Redis: SET NX agent_builder:jti:<jti_str> "consumed" EX 86400  （24h TTL）
- Postgres: HitlToken.used_at 是真实权威
- check(jti): Redis 命中 → 直接 True；miss → 查 Postgres → 命中则回填 Redis
- consume(jti, ip, ua): 事务内 UPDATE...WHERE used_at IS NULL RETURNING * + Redis pipeline 设置
"""

from __future__ import annotations
import logging
from uuid import UUID
from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update

from app.agent_builder.models.hitl_token import HitlToken

log = logging.getLogger(__name__)
_REDIS_PREFIX = "agent_builder:jti:"
_REDIS_TTL = 86400  # 24h

class HitlTokenStore:
    def __init__(self, db: AsyncSession, redis: Redis):
        self.db = db
        self.redis = redis

    async def is_consumed(self, jti: UUID) -> bool:
        """检查 jti 是否已消费（Redis-first，miss 回查 PG）"""
        key = f"{_REDIS_PREFIX}{jti}"
        if await self.redis.exists(key):
            return True
        # Redis miss，查 Postgres
        result = await self.db.execute(
            select(HitlToken.used_at).where(HitlToken.jti == jti)
        )
        used_at = result.scalar_one_or_none()
        if used_at is not None:
            await self.redis.set(key, "consumed", ex=_REDIS_TTL)
            return True
        return False

    async def consume(self, jti: UUID, ip: str | None, ua: str | None) -> HitlToken | None:
        """原子消费 jti：UPDATE...WHERE used_at IS NULL RETURNING *

        Returns:
            HitlToken if successfully consumed (zero-row return = already consumed → None)
        """
        from datetime import datetime, timezone
        stmt = (
            update(HitlToken)
            .where(HitlToken.jti == jti, HitlToken.used_at.is_(None))
            .values(used_at=datetime.now(timezone.utc), used_ip=ip, used_ua=ua)
            .returning(HitlToken)
        )
        result = await self.db.execute(stmt)
        row = result.scalar_one_or_none()
        if row is not None:
            await self.redis.set(f"{_REDIS_PREFIX}{jti}", "consumed", ex=_REDIS_TTL)
        return row

    async def invalidate_siblings(self, node_state_id: UUID, except_jti: UUID) -> int:
        """同 node_state_id 下其他未消费 token 同时失效（HITL-01 防止重复决策）"""
        from datetime import datetime, timezone
        stmt = (
            update(HitlToken)
            .where(
                HitlToken.node_state_id == node_state_id,
                HitlToken.jti != except_jti,
                HitlToken.used_at.is_(None),
            )
            .values(used_at=datetime.now(timezone.utc), used_ip="system:sibling-invalidate")
            .returning(HitlToken.jti)
        )
        result = await self.db.execute(stmt)
        invalidated = [str(jti) for jti in result.scalars().all()]
        if invalidated:
            pipe = self.redis.pipeline()
            for jti_str in invalidated:
                pipe.set(f"{_REDIS_PREFIX}{jti_str}", "consumed", ex=_REDIS_TTL)
            await pipe.execute()
        return len(invalidated)
```
