```python
# backend/app/services/hitl_token_service.py
"""HITL Token JWT 服务 — HS256 签发 + 解码（aud='hitl'）。

设计参考 docs/reading-dify-03-03-hitl-token-2026-05-17.md §7 GET 不消费 jti 契约。
- 复用 Phase 1 jwt_service._get_jwt_secret()（HMAC_SECRET 启动校验 ≥ 32）
- aud='hitl' 严格与 session/email-verify/invite 区分
- decode 仅校签 + exp + aud；不消费 jti（jti 消费在 HitlTokenStore.consume，由 03-06 调用）
"""

from __future__ import annotations
import logging
from datetime import datetime, timezone
from typing import Literal
from uuid import UUID

import jwt as pyjwt

from app.services.jwt_service import _get_jwt_secret  # 复用 Phase 1

log = logging.getLogger(__name__)

JWT_AUD = "hitl"
JWT_ISS = "agent_builder"
JWT_ALG = "HS256"


class HitlTokenError(Exception):
    """HITL Token 校验异常基类。"""


class InvalidSignature(HitlTokenError):
    """签名错误。"""


class TokenExpired(HitlTokenError):
    """token 过期。"""


class InvalidAudience(HitlTokenError):
    """audience 不匹配（防 cross-token 滥用）。"""


class HitlTokenService:
    @staticmethod
    def sign(
        *,
        jti: UUID,
        instance_id: UUID,
        node_state_id: UUID,
        actor_id: UUID,
        action: Literal["submit", "approve", "return", "reject"],
        role: Literal["executor", "reviewer"],
        expires_at: datetime,
    ) -> str:
        """签发 HITL token（HS256）。"""
        secret = _get_jwt_secret()
        payload = {
            "iss": JWT_ISS,
            "aud": JWT_AUD,
            "iat": int(datetime.now(timezone.utc).timestamp()),
            "exp": int(expires_at.timestamp()),
            "jti": str(jti),
            "flow_id": str(instance_id),
            "node_state_id": str(node_state_id),
            "actor_id": str(actor_id),
            "role": role,
            "allowed_actions": [action],
        }
        return pyjwt.encode(payload, secret, algorithm=JWT_ALG)

    @staticmethod
    def decode(token: str) -> dict:
        """解码 + 校签 + exp + aud。不消费 jti。"""
        secret = _get_jwt_secret()
        try:
            payload = pyjwt.decode(
                token,
                secret,
                algorithms=[JWT_ALG],
                audience=JWT_AUD,
                issuer=JWT_ISS,
                options={"require": ["jti", "exp", "iat", "aud", "iss"]},
            )
        except pyjwt.ExpiredSignatureError as e:
            raise TokenExpired(str(e)) from e
        except pyjwt.InvalidAudienceError as e:
            raise InvalidAudience(str(e)) from e
        except pyjwt.InvalidSignatureError as e:
            raise InvalidSignature(str(e)) from e
        except pyjwt.InvalidTokenError as e:
            raise HitlTokenError(str(e)) from e
        return payload
```

```python
# backend/app/agent_builder/security/bot_detector.py
"""Safe Links Bot UA 检测（Pitfall 3 P0 防护）。

设计参考 docs/reading-dify-03-03-hitl-token-2026-05-17.md。
CLAUDE.md 2.5：GET /hitl/page/<token> 检测到 bot UA 直接返回静态 HTML，不签 session cookie。
"""

from __future__ import annotations

# CONTEXT.md §Safe Links bot UA 白名单（13 个）
BOT_UA_PATTERNS: tuple[str, ...] = (
    "microsoftdefender",
    "outlook-safelinks",
    "safelinks",  # 通用前缀
    "slackbot-linkexpanding",
    "twitterbot",
    "facebookexternalhit",
    "linkedinbot",
    "whatsapp",
    "googlebot",
    "telegrambot",
    "discordbot",
    "duckduckbot",
    "baiduspider",
    "bingbot",
    "ac-detector-tool",  # CONTEXT.md §Specific Ideas Outlook 真实 UA
)


def is_bot_ua(ua: str | None) -> bool:
    """检测 User-Agent 是否为已知邮件扫描器 / link preview bot。

    Case-insensitive substring 匹配。None / 空 string 安全返回 False。
    """
    if not ua:
        return False
    ua_lower = ua.lower()
    return any(pattern in ua_lower for pattern in BOT_UA_PATTERNS)
```
