```python
# backend/app/agent_builder/workflow/nodes/notification.py
from __future__ import annotations
import logging
from typing import Any
from uuid import UUID, uuid4

from app.agent_builder.workflow.nodes.base import BaseNodeExecutor

log = logging.getLogger(__name__)


class NotificationNodeExecutor(BaseNodeExecutor):
    """独立通知节点（NODE-07）— 不暂停 graph，发送后继续。

    设计参考 reading-dify-03-05-notification-node-2026-05-17.md §7：
    - 与 HITL 节点的核心区别：不创建 hitl_token、不调用 interrupt()
    - 失败不阻断 graph：单封失败 → 记录 failed_count + state；graph 继续
    """

    async def execute(self, config: dict, state: dict) -> dict:
        """渲染 recipients/subject/body + 入队 + 返回统计。"""
        from app.services.notification_service import NotificationService
        from app.agent_builder.workflow.checkpoint import parse_thread_id

        channels = config.get("channels", ["email"])
        if "email" not in channels:
            log.warning("Phase 3 仅支持 email channel；忽略其它")
            return {self.node_id: {"sent_count": 0, "failed_count": 0, "skipped": True}}

        recipients = config.get("recipients", [])
        if isinstance(recipients, str):
            recipients = [recipients]
        recipients = [r for r in recipients if r and "@" in r]

        # subject + body 已经在 _render_config 中渲染（BaseNodeExecutor）
        subject = config.get("subject", "通知")
        body = config.get("body", "")

        # 从 state injected context 取 ws/instance/node_state
        # (BaseNodeExecutor 注入 self._workspace_id / instance_id / node_state_id 由 ExecutionEngine)
        workspace_id = self._workspace_id or UUID(state.get("__workspace_id"))
        instance_id = self._instance_id or UUID(state.get("__instance_id"))
        node_state_id = UUID(state.get("__node_state_id"))

        sent_count = 0
        failed_count = 0
        notification_ids: list[int] = []

        # 用 ExecutionEngine 提供的 db / arq pool
        db = self._db
        svc = NotificationService(db=db, arq_pool=self._arq_pool)
        for recipient in recipients:
            try:
                notif = await svc.enqueue_generic_email(
                    workspace_id=workspace_id,
                    instance_id=instance_id,
                    node_state_id=node_state_id,
                    recipient_email=recipient,
                    subject=subject,
                    body=body,
                )
                notification_ids.append(notif.id)
                sent_count += 1
            except Exception as e:
                log.exception(f"notification 入队失败 recipient={recipient}: {e}")
                failed_count += 1

        return {
            self.node_id: {
                "sent_count": sent_count,
                "failed_count": failed_count,
                "notification_ids": notification_ids,
            }
        }
```
