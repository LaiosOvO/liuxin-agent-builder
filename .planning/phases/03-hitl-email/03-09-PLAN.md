# backend/app/jobs/hitl_timeout_jobs.py
"""HITL 节点超时催办 + 升级 — arq cron job。

设计参考 reading-dify-03-09-timeout-worker-2026-05-17.md §7：
- arq cron(60s) 替代 Celery beat
- notifications UNIQUE 约束 防多 worker 重发（03-01 已建）
- advisory_lock(node_state_id) 处理 race condition
"""

from __future__ import annotations
import logging
from datetime import datetime, timezone, timedelta

from sqlalchemy import select, text
from sqlalchemy.ext.asyncio import AsyncSession

from app.agent_builder.models.node_state import NodeState
from app.agent_builder.models.notification import Notification

log = logging.getLogger(__name__)

# 阶梯阈值（秒）
_ROUND_1_THRESHOLD = 24 * 3600    # 24h
_ROUND_2_THRESHOLD = 48 * 3600    # 48h
_ESCALATE_THRESHOLD = 72 * 3600   # 72h


async def scan_hitl_timeouts(ctx) -> int:
    """arq cron entry — 每 60s 扫一次。

    Returns:
        本次处理的节点数。
    """
    db: AsyncSession = ctx["db"]
    redis = ctx["redis"]
    arq = ctx["arq_pool"]

    now = datetime.now(timezone.utc)

    # 1. 查所有可能超时的 HITL 节点
    result = await db.execute(
        select(NodeState).where(
            NodeState.status.in_(["waiting_human", "in_review"]),
            NodeState.payload["deadline_at"].astext.cast(sa.DateTime) < now,
        )
    )
    nodes = result.scalars().all()
    processed = 0

    for ns in nodes:
        # 2. advisory_lock 防 race
        lock_key = hash(str(ns.id)) & 0x7FFFFFFFFFFFFFFF
        async with db.begin_nested():
            await db.execute(text("SELECT pg_advisory_xact_lock(:k)"), {"k": lock_key})

            # 3. 算超时时长（相对 started_at）
            started_at = datetime.fromisoformat(ns.payload["started_at"])
            elapsed = (now - started_at).total_seconds()

            # 4. 查当前最大 reminder_round
            max_round_result = await db.execute(
                select(sa.func.max(Notification.reminder_round)).where(
                    Notification.node_state_id == ns.id,
                    Notification.channel == "email",
                )
            )
            current_round = max_round_result.scalar_one_or_none() or 0

            # 5. 决定下一步
            if elapsed >= _ESCALATE_THRESHOLD and current_round < 3:
                await _trigger_escalation(db, redis, arq, ns)
            elif elapsed >= _ROUND_2_THRESHOLD and current_round < 2:
                await _trigger_reminder(db, redis, arq, ns, round=2)
            elif elapsed >= _ROUND_1_THRESHOLD and current_round < 1:
                await _trigger_reminder(db, redis, arq, ns, round=1)
            else:
                continue
            processed += 1

    return processed


async def _trigger_reminder(db, redis, arq, ns: NodeState, *, round: int) -> None:
    """入队同模板邮件，reminder_round=round（复用 03-04 send_hitl_email_job）。"""
    from app.services.notification_service import NotificationService
    svc = NotificationService(db, arq)
    # ... 用 ns.payload 解析 actor email 等
    actor = ns.payload["current_actor"]
    # 复用 batch_create_tokens 重新签发 token（前面 token 已过期）
    from app.agent_builder.services.hitl_service import HitlService
    hitl_svc = HitlService(db)
    new_tokens = await hitl_svc.batch_create_tokens(
        instance_id=ns.instance_id,
        node_state_id=ns.id,
        actor_id=actor["id"],
        allowed_actions=["submit", "return", "reject"] if ns.payload["phase"] == "submit" else ["approve", "return", "reject"],
    )
    await svc.enqueue_hitl_email(
        workspace_id=ns.workspace_id,
        instance_id=ns.instance_id,
        node_state_id=ns.id,
        recipient_email=actor["email"],
        tokens=new_tokens,
        form_schema=ns.payload.get("form_schema", {}),
        deadline_at=ns.payload["deadline_at"],
        actor_name=actor.get("email", ""),
        flow_title="...",
        node_title="...",
        applicant_name="...",
        description=ns.payload.get("description", ""),
        reminder_round=round,
    )


async def _trigger_escalation(db, redis, arq, ns: NodeState) -> None:
    """升级到 escalate_to：写 records + 发升级邮件给新 actor + audit。"""
    from app.agent_builder.services.escalation_service import EscalationService
    es = EscalationService(db, arq)
    await es.perform_escalation(ns)
