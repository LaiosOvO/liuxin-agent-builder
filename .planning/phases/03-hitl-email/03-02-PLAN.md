```python
# backend/app/agent_builder/workflow/nodes/hitl.py
"""HITL 节点 executor — LangGraph interrupt + Command(resume) 集成。

设计参考 docs/reading-dify-03-02-hitl-executor-2026-05-17.md：
- Dify human_input_adapter EmailDeliveryConfig 模式 → 我们用 node_def.config.email_recipients
- LangGraph 1.2 interrupt 返回值在 resume 后注入，节点函数前后两段执行

不能复用 BaseNodeExecutor.__call__ 的重试装饰器，因为：
- interrupt 会 raise GraphInterrupt 跳出 ainvoke，外层 graph runner 会 ainvoke(Command(resume=...))
- 重试装饰器会把 interrupt 当 normal exception 吞掉
"""

from __future__ import annotations
import logging
from datetime import datetime, timezone
from typing import Any
from uuid import UUID

from langgraph.types import interrupt

from app.agent_builder.workflow.nodes.base import BaseNodeExecutor, NodeExecutionError
from app.agent_builder.workflow.hitl_payload import (
    build_initial_payload,
    append_record,
    compute_next_status,
    validate_form_data,
)

log = logging.getLogger(__name__)


class HITLNodeExecutor(BaseNodeExecutor):
    """HITL 节点：四态决策（HITL-01）+ 单 interrupt + 自管审批链状态（HITL-03）。

    v1（Phase 3）支持 single 模式（mode=single, 1 个 approver）；
    sequential / parallel_all / parallel_any → Phase 4。
    """

    async def __call__(self, state: dict) -> dict:
        """LangGraph node fn 入口。本 executor override 基类 __call__ 因为 interrupt 不能被重试装饰器包裹。"""
        # 1. Pre-interrupt: 初始化 payload + 创建 tokens + 入队 notification
        rendered_config = self._render_config(state)
        node_state_id = state.get("__node_state_id")  # 由 ExecutionEngine 注入
        if not node_state_id:
            raise NodeExecutionError(self.node_id, "缺少 __node_state_id")

        # 2. 节点 enter logic（仅首次执行，resume 后此段不再走 — interrupt 已 cache）
        # interrupt 是 1.2 的"暂停信号"，第二次 ainvoke(Command(resume=...)) 后 interrupt() 直接返回 resume value
        decision = interrupt({
            "node_state_id": str(node_state_id),
            "phase": "submit",  # Phase 3 single 模式：submit 直接到 in_review，approve = done
            "form_schema": rendered_config.get("form_schema", {}),
            "deadline_at": rendered_config.get("deadline_at"),  # 由 ExecutionEngine pre-fill
        })

        # 3. Resume 后逻辑：decision = {action, reason, form_data, actor_id, ip, ua, jti}
        log.info(f"HITL node {self.node_id} resumed with action={decision.get('action')}")

        # 4. form_schema 校验 form_data
        form_schema = rendered_config.get("form_schema", {})
        if form_schema and decision.get("form_data"):
            validate_form_data(form_schema, decision["form_data"])

        # 5. Return result 写入 state（merge）
        return {
            self.node_id: {
                "action": decision["action"],
                "reason": decision.get("reason", ""),
                "form_data": decision.get("form_data", {}),
                "actor_id": decision.get("actor_id"),
                "completed_at": datetime.now(timezone.utc).isoformat(),
            }
        }
```

```python
# backend/app/agent_builder/workflow/hitl_payload.py
"""HITL payload 纯函数（无副作用，便于单测）。

设计参考 docs/reading-dify-03-02-hitl-executor-2026-05-17.md §7。
"""

from __future__ import annotations
from datetime import datetime, timezone, timedelta
from typing import Any, Literal
from uuid import UUID

from jsonschema import Draft7Validator, ValidationError as JsonSchemaValidationError

NodeStatus = Literal["pending", "waiting_human", "in_review", "done", "rejected", "returned"]
Action = Literal["submit", "approve", "return", "reject"]


def build_initial_payload(
    *,
    current_actor_id: UUID,
    current_actor_email: str,
    role: Literal["executor", "reviewer"],
    timeout_seconds: int,
    form_schema: dict,
    approvers: list[UUID] | None = None,
) -> dict[str, Any]:
    """构造 HITL 节点初始 payload（写入 node_state.payload）。"""
    now = datetime.now(timezone.utc)
    return {
        "phase": "submit" if role == "executor" else "review",
        "current_actor": {
            "id": str(current_actor_id),
            "email": current_actor_email,
            "role": role,
        },
        "approval_chain": {
            "mode": "single",  # Phase 3 only single
            "approvers": [str(a) for a in (approvers or [current_actor_id])],
            "current_idx": 0,
        },
        "records": [],
        "pending_approvers": [],
        "started_at": now.isoformat(),
        "deadline_at": (now + timedelta(seconds=timeout_seconds)).isoformat(),
        "form_schema": form_schema,
    }


def append_record(payload: dict, *, actor_id: UUID, actor_email: str, action: Action,
                  reason: str = "", form_data: dict | None = None,
                  ip: str | None = None, ua: str | None = None) -> dict:
    """返回新 payload（immutable，不修改原 dict）+ append 一条 record。"""
    new_payload = {**payload}
    new_payload["records"] = [
        *payload.get("records", []),
        {
            "actor_id": str(actor_id),
            "actor_email": actor_email,
            "action": action,
            "reason": reason,
            "form_data": form_data or {},
            "ts": datetime.now(timezone.utc).isoformat(),
            "ip": ip,
            "ua": ua,
        },
    ]
    return new_payload


def compute_next_status(action: Action, current_phase: Literal["submit", "review"]) -> NodeStatus:
    """根据 action 计算节点新 status。

    Phase 3 single 模式：
    - submit phase + action=submit → in_review（单人模式下直接转 review）
    - review phase + action=approve → done
    - any phase + action=return → returned
    - any phase + action=reject → rejected
    """
    if action == "return":
        return "returned"
    if action == "reject":
        return "rejected"
    if action == "submit" and current_phase == "submit":
        return "in_review"
    if action == "approve" and current_phase == "review":
        return "done"
    raise ValueError(f"非法 action={action} 在 phase={current_phase}")


def validate_form_data(schema: dict, data: dict) -> None:
    """用 jsonschema 校验 form_data；失败抛 JsonSchemaValidationError。"""
    Draft7Validator(schema).validate(data)
```

```python
# backend/app/agent_builder/services/hitl_service.py
from __future__ import annotations
from datetime import datetime, timezone, timedelta
from uuid import UUID, uuid4

from sqlalchemy.ext.asyncio import AsyncSession

from app.agent_builder.models.hitl_token import HitlToken
from app.agent_builder.models.node_state import NodeState

class HitlService:
    """HITL 业务逻辑层（被 HITLNodeExecutor + 03-06 API 复用）。"""

    def __init__(self, db: AsyncSession):
        self.db = db

    async def batch_create_tokens(
        self,
        *,
        instance_id: UUID,
        node_state_id: UUID,
        actor_id: UUID,
        allowed_actions: list[str],
        expires_in_seconds: int = 86400,
    ) -> list[HitlToken]:
        """为 actor 批量生成 token（每个 allowed_action 一个 jti）。"""
        expires_at = datetime.now(timezone.utc) + timedelta(seconds=expires_in_seconds)
        tokens = [
            HitlToken(
                jti=uuid4(),
                instance_id=instance_id,
                node_state_id=node_state_id,
                actor_id=actor_id,
                action=action,
                expires_at=expires_at,
            )
            for action in allowed_actions
        ]
        self.db.add_all(tokens)
        await self.db.flush()  # 让 jti 生成，但不 commit（外层事务管理）
        return tokens

    async def resolve_allowed_actions(self, phase: str) -> list[str]:
        """Phase=submit → [submit, return, reject]；phase=review → [approve, return, reject]"""
        if phase == "submit":
            return ["submit", "return", "reject"]
        if phase == "review":
            return ["approve", "return", "reject"]
        raise ValueError(f"非法 phase={phase}")
```
