```tsx
// web/src/components/hitl/decision-form.tsx
"use client";
import { useState } from "react";
import Form from "@rjsf/core";
import validator from "@rjsf/validator-ajv8";
import type { RJSFSchema } from "@rjsf/utils";
import { submitHitlAction } from "@/lib/api/hitl";

interface DecisionFormProps {
  token: string;
  jti: string;
  formSchema: RJSFSchema;
  phase: "submit" | "review";
}

const ACTIONS_BY_PHASE: Record<string, { id: string; label: string; color: string }[]> = {
  submit: [
    { id: "submit", label: "提交", color: "bg-emerald-600 hover:bg-emerald-700" },
    { id: "return", label: "退回", color: "bg-amber-500 hover:bg-amber-600" },
    { id: "reject", label: "拒绝", color: "bg-red-500 hover:bg-red-600" },
  ],
  review: [
    { id: "approve", label: "通过", color: "bg-emerald-600 hover:bg-emerald-700" },
    { id: "return", label: "退回", color: "bg-amber-500 hover:bg-amber-600" },
    { id: "reject", label: "拒绝", color: "bg-red-500 hover:bg-red-600" },
  ],
};

export function DecisionForm({ token, jti, formSchema, phase }: DecisionFormProps) {
  const [formData, setFormData] = useState<Record<string, unknown>>({});
  const [reason, setReason] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const actions = ACTIONS_BY_PHASE[phase];

  async function handleSubmit(action: string) {
    if (submitting) return;
    setSubmitting(true);
    setError(null);
    try {
      const result = await submitHitlAction(token, { action, reason, form_data: formData });
      if (result.ok) {
        window.location.href = `/hitl/success/${result.instance_id}`;
      } else {
        setError(result.error || "提交失败");
        setSubmitting(false);
      }
    } catch (e) {
      setError(String(e));
      setSubmitting(false);
    }
  }

  return (
    <div className="space-y-6">
      {/* JSON Schema → 表单 */}
      <Form
        schema={formSchema}
        validator={validator}
        formData={formData}
        onChange={(e) => setFormData(e.formData)}
        // 关闭 RJSF 自带 submit 按钮，自己控制
        uiSchema={{ "ui:submitButtonOptions": { norender: true } }}
      >
        <></>
      </Form>

      {/* Reason */}
      <div>
        <label className="block text-sm font-medium mb-2">备注 / 退回理由</label>
        <textarea
          className="w-full border rounded-md p-2 min-h-[80px]"
          value={reason}
          onChange={(e) => setReason(e.target.value)}
          placeholder="可选；退回 / 拒绝建议填写理由"
        />
      </div>

      {error && <div className="text-red-600 text-sm bg-red-50 p-3 rounded">{error}</div>}

      {/* Action 按钮 */}
      <div className="flex gap-3 flex-wrap">
        {actions.map((a) => (
          <button
            key={a.id}
            type="button"
            disabled={submitting}
            onClick={() => handleSubmit(a.id)}
            className={`${a.color} text-white px-6 py-2 rounded-md disabled:opacity-50 disabled:cursor-not-allowed`}
          >
            {submitting ? "处理中..." : a.label}
          </button>
        ))}
      </div>
    </div>
  );
}
```
