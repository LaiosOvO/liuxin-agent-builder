```python
"""通知服务 — HITL 邮件 + IM（Phase 4+）的入队 + 状态管理。

设计参考 reading-dify-03-04-email-delivery：
- 简化 Dify 三层 ORM 为 notifications 单表 + 入队 arq
"""

from __future__ import annotations
import logging
from datetime import datetime
from typing import Sequence
from uuid import UUID

from sqlalchemy.ext.asyncio import AsyncSession

from app.agent_builder.models.notification import Notification
from app.agent_builder.models.hitl_token import HitlToken

log = logging.getLogger(__name__)

class NotificationService:
    def __init__(self, db: AsyncSession, arq_pool=None):
        self.db = db
        self.arq = arq_pool

    async def enqueue_hitl_email(
        self,
        *,
        workspace_id: UUID,
        instance_id: UUID,
        node_state_id: UUID,
        recipient_email: str,
        tokens: Sequence[HitlToken],
        form_schema: dict,
        deadline_at: datetime,
        actor_name: str,
        flow_title: str,
        node_title: str,
        applicant_name: str,
        description: str,
        reminder_round: int = 0,  # 0=首次发送
    ) -> Notification:
        """写 notifications.status=pending + 入队 arq send_hitl_email_job。

        Returns:
            创建的 Notification 行。
        """
        notif = Notification(
            workspace_id=workspace_id,
            instance_id=instance_id,
            node_state_id=node_state_id,
            channel="email",
            recipient=recipient_email,
            reminder_round=reminder_round,
            status="pending",
            payload={
                "tokens": [{"jti": str(t.jti), "action": t.action} for t in tokens],
                "form_schema": form_schema,
                "deadline_at": deadline_at.isoformat(),
                "actor_name": actor_name,
                "flow_title": flow_title,
                "node_title": node_title,
                "applicant_name": applicant_name,
                "description": description,
            },
        )
        self.db.add(notif)
        await self.db.flush()
        await self.db.commit()

        if self.arq:
            await self.arq.enqueue_job("send_hitl_email_job", str(notif.id))
        else:
            # fallback: asyncio.create_task（测试 / dev）
            from app.jobs.email_jobs import send_hitl_email_job
            import asyncio
            asyncio.create_task(send_hitl_email_job(None, str(notif.id)))
        return notif

    async def mark_sent(self, notification_id: int) -> None:
        ...

    async def mark_failed(self, notification_id: int, error: str) -> None:
        ...
```

```python
"""arq job: 发送 HITL 邮件（NOTI-10 重试）。

设计参考 reading-dify-03-04-email-delivery §7。
"""

from __future__ import annotations
import logging
import os
from datetime import datetime

import aiosmtplib
from tenacity import AsyncRetrying, RetryError, stop_after_attempt, wait_exponential, retry_if_exception_type

from app.services.email_service import _send_email  # Phase 1 底层
from jinja2 import Environment, FileSystemLoader, select_autoescape
from pathlib import Path

log = logging.getLogger(__name__)

_TEMPLATE_DIR = Path(__file__).parent.parent / "templates" / "email"
_jinja_env: Environment | None = None

def _get_jinja_env() -> Environment:
    global _jinja_env
    if _jinja_env is None:
        _jinja_env = Environment(
            loader=FileSystemLoader(str(_TEMPLATE_DIR)),
            autoescape=select_autoescape(["html", "xml"]),
        )
    return _jinja_env

def _build_deeplink(jti: str) -> str:
    base = os.environ.get("PUBLIC_BASE_URL", "http://localhost:8000").rstrip("/")
    return f"{base}/hitl/page/{jti}"

async def send_hitl_email_job(ctx, notification_id: str) -> None:
    """arq job entrypoint. ctx 是 arq context dict（worker 注入 db / redis）。"""
    from sqlalchemy.ext.asyncio import AsyncSession
    db: AsyncSession = ctx["db"] if ctx else _get_dev_db()  # dev fallback
    from app.agent_builder.models.notification import Notification
    notif = await db.get(Notification, int(notification_id))
    if not notif:
        log.error(f"notification {notification_id} not found")
        return
    if notif.status == "sent":
        log.info(f"notification {notification_id} already sent, skip (idempotent)")
        return
    notif.status = "sending"
    await db.commit()

    env = _get_jinja_env()
    template = "hitl_reminder.html" if notif.reminder_round > 0 else "hitl_decision.html"
    html = env.get_template(template).render(
        **notif.payload,
        deeplinks=[
            {**t, "url": _build_deeplink(t["jti"])}
            for t in notif.payload["tokens"]
        ],
    )
    text = env.get_template("hitl_decision_text.txt").render(
        **notif.payload,
        deeplinks=[
            {**t, "url": _build_deeplink(t["jti"])}
            for t in notif.payload["tokens"]
        ],
    )
    subject_prefix = "[催办] " if notif.reminder_round > 0 else ""
    subject = f"{subject_prefix}审批待办：{notif.payload['flow_title']} - {notif.payload['node_title']}"

    try:
        async for attempt in AsyncRetrying(
            stop=stop_after_attempt(3),
            wait=wait_exponential(multiplier=1, min=1, max=4),
            retry=retry_if_exception_type((aiosmtplib.errors.SMTPException, ConnectionError, TimeoutError)),
            reraise=True,
        ):
            with attempt:
                await _send_email(to=notif.recipient, subject=subject, html_body=html, text_body=text)
        notif.status = "sent"
        from datetime import datetime, timezone
        notif.sent_at = datetime.now(timezone.utc)
    except Exception as e:
        notif.status = "failed"
        notif.error_message = str(e)[:1000]
        log.exception(f"failed to send HITL email after retries: notification_id={notification_id}")
    finally:
        await db.commit()
```
