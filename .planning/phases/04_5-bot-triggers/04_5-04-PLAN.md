```python
class BotCommand:
    name: str
    args: dict[str, str]
    self_apply: bool = False
    raw: str = ""


def parse_command(raw: str, bot_config: BotConfig) -> BotCommand | None:
    """5 步解析：
    1) 剥 @<bot_username> 前缀
    2) 命中 self_apply.sentinel_phrases → 立即返回 BotCommand(name=cmd, self_apply=True)
    3) tokenize（首 token = 命令名）
    4) 白名单查 BotConfig.commands（不在白名单 → 返回 None）
    5) 按 CommandSpec.args 正则校验 — 失败 raise BotArgError（friendly message + 用法示例）
    """
```

       ```python
       hr/offboarding-flow 无此模块（v1 简化）；本项目复用 Phase 3 HitlTokenStore 模式。

       import logging
       from datetime import datetime, UTC

       from redis.asyncio import Redis

       log = logging.getLogger(__name__)


       class RateLimiter:
           def __init__(self, redis: Redis, *, per_user_per_minute: int = 10) -> None:
               self._redis = redis
               self._limit = per_user_per_minute

           async def check_and_increment(
               self, *, workspace_id: str, bot_name: str, sender_user_id: str
           ) -> bool:
               bucket = datetime.now(UTC).strftime("%Y%m%d%H%M")
               key = f"bot:rl:{workspace_id}:{bot_name}:{sender_user_id}:{bucket}"
               # INCR + EXPIRE NX (仅首次设过期)
               pipe = self._redis.pipeline()
               pipe.incr(key)
               pipe.expire(key, 70, nx=True)  # 70s 给 jitter；fixed-window

               current = results[0]

                       "bot.dispatch.rate_limited",

                           "workspace_id": workspace_id,
                           "bot_name": bot_name,
                           "sender_user_id": sender_user_id,
                           "current": current,
                           "limit": self._limit,
                       },
                   )
                   return False
               return True
       ```

       ```python
       """idempotency: sha256(ws + bot + user + cmd + canonical_args + minute_bucket) → Redis SET NX。

       防 Mattermost at-least-once 重推 + 用户秒重发同样命令。
       """
       from __future__ import annotations
       import hashlib
       import json
       from datetime import datetime, UTC

       from redis.asyncio import Redis


       def compute_idempotency_key(
           *,
           workspace_id: str,
           bot_name: str,
           sender_user_id: str,
           cmd_name: str,
           args: dict[str, str],
           bucket_minutes: int = 1,
       ) -> str:
           bucket = datetime.now(UTC).strftime("%Y%m%d%H%M")
           payload = json.dumps(
               {
                   "ws": workspace_id,
                   "bot": bot_name,
                   "user": sender_user_id,
                   "cmd": cmd_name,
                   "args": dict(sorted(args.items())),
                   "bucket": bucket,
               },
               sort_keys=True,
               ensure_ascii=False,
           )
           digest = hashlib.sha256(payload.encode("utf-8")).hexdigest()[:16]
           return f"bot:idem:{digest}"


       class IdempotencyStore:
           def __init__(self, redis: Redis) -> None:
               self._redis = redis

           async def check_and_set(self, key: str, ttl_seconds: int = 120) -> bool:
               result = await self._redis.set(key, "1", nx=True, ex=ttl_seconds)
               return result is not None


       __all__ = ["compute_idempotency_key", "IdempotencyStore"]
       ```

    ```python
    设计要点（用户决策 + RESEARCH §Architecture Patterns）:
    - 命令冲突解决顺序硬性: keywords self_apply > parse_command 白名单 > LLM intent router > ai_qa 兜底（Pitfall 11）
    - 异常 wrap: BotError 走 user_message 友好提示；其他 raise wrap 为 BotInternalError("❌ 内部错误（事件 ID: <audit_id>）") - N-IM-04 + Pitfall 4

    - 身份对齐: sender_name → users 表查 + workspace_id 边界 → 失败按 on_unknown 决策（R-IM-06）
    - 结构化日志: bot.dispatch.{entered, routed_via, completed, failed} 含 bot_name+workspace_id+sender+cmd+latency_ms+outcome（Phase 7 Run Viewer）
    """
    from __future__ import annotations
    import logging
    import time
    from collections.abc import Awaitable, Callable
    from dataclasses import replace
    from uuid import UUID

    from sqlalchemy import select
    from sqlalchemy.ext.asyncio import AsyncSession

    from app.agent_builder.models.user import User
    from .audit import AuditLogger
    from .context import BotContext, IMHelpers
    from .exceptions import (
        BotArgError,
        BotError,
        BotIdentityError,
        BotInternalError,
        BotPermissionError,
        BotRateLimitError,
    )
    from .idempotency import IdempotencyStore, compute_idempotency_key
    from .llm_router import BotIntentRouter, IntentResult
    from .parser import BotCommand, parse_command
    from .rate_limiter import RateLimiter
    from .registry import HandlerRegistry
    from .schemas.bot_config import BotConfig

    log = logging.getLogger(__name__)


    class BotDispatcher:
        def __init__(
            self,
            *,
            bot_config: BotConfig,
            registry: HandlerRegistry,
            intent_router: BotIntentRouter,
            rate_limiter: RateLimiter,
            idempotency: IdempotencyStore,
            audit: AuditLogger,
            session_factory: Callable[[], AsyncSession],
        ) -> None:
            self.bot_config = bot_config
            self.registry = registry
            self.intent_router = intent_router
            self.rate_limiter = rate_limiter
            self.idempotency = idempotency
            self.audit = audit
            self._session_factory = session_factory

        async def dispatch_message(
            self,
            *,
            sender_name: str,
            sender_im_user_id: str,
            channel_id: str,
            channel_type: str,
            message: str,
            workspace_id: UUID,
            im_helpers: IMHelpers,
        ) -> None:
            t0 = time.monotonic()
            log.info(
                "bot.dispatch.entered",
                extra={
                    "bot_name": self.bot_config.name,
                    "workspace_id": str(workspace_id),
                    "sender_name": sender_name,
                    "channel_type": channel_type,
                },
            )

            # 占位 ctx — 身份对齐前
            ctx_pre = BotContext(
                workspace_id=str(workspace_id),
                bot_config=self.bot_config,
                user_name=sender_name,
                user_id=None,
                user_role=None,
                user_email=None,
                channel_id=channel_id,
                channel_type=channel_type,
                sender_im_user_id=sender_im_user_id,
                raw_message=message,
                im_helpers=im_helpers,
            )

            try:
                # 1. 身份对齐（R-IM-06）
                ctx = await self._align_identity(sender_name, ctx_pre)
                if ctx is None:
                    # on_unknown=reject_friendly 已 post 提示；audit + return
                    await self._audit_outcome(
                        ctx_pre, None, None, "permission_denied",
                        "unknown user", time.monotonic() - t0, "whitelist", None,
                    )
                    return

                # 2. Rate limit 闸门（R-IM-12）
                allowed = await self.rate_limiter.check_and_increment(
                    workspace_id=str(workspace_id),
                    bot_name=self.bot_config.name,
                    sender_user_id=str(ctx.user_id),
                )
                if not allowed:
                    await im_helpers.post_channel(channel_id, BotRateLimitError().user_message)
                    await self._audit_outcome(
                        ctx, None, None, "rate_limited", None,
                        time.monotonic() - t0, "whitelist", None,
                    )
                    return

                # 3. 命令路由 — 三层硬性顺序（Pitfall 11）
                routed_via, cmd, intent_result = await self._route_command(message, ctx)

                # 4. Idempotency（路由确定后才计算 — 用 cmd_name + args）
                cmd_name = cmd.name if cmd else (intent_result.intent if intent_result else "ai_qa")
                cmd_args = cmd.args if cmd else (intent_result.args if intent_result else {})
                idem_key = compute_idempotency_key(
                    workspace_id=str(workspace_id),
                    bot_name=self.bot_config.name,
                    sender_user_id=str(ctx.user_id),
                    cmd_name=cmd_name,
                    args=cmd_args,
                )
                is_first = await self.idempotency.check_and_set(idem_key, ttl_seconds=120)
                if not is_first:
                    log.info(
                        "bot.dispatch.duplicate_skipped",
                        extra={"idem_key": idem_key, "cmd_name": cmd_name},
                    )
                    # 不回帖（防 IM 重推时刷屏）；audit 记录
                    await self._audit_outcome(
                        ctx, cmd_name, cmd_args, "success", "duplicate_skipped",
                        time.monotonic() - t0, routed_via, intent_result.confidence if intent_result else None,
                    )
                    return

                # 5. 执行
                if cmd is not None:
                    response = await self.registry.dispatch(cmd.name, cmd.args, ctx)
                elif intent_result is not None and intent_result.intent == "ai_qa":
                    response = await self.intent_router.answer_ai_qa(message=message, ctx=ctx)
                elif intent_result is not None:
                    # confidence >= threshold + intent 是命令 — 通过 registry dispatch
                    response = await self.registry.dispatch(intent_result.intent, intent_result.args, ctx)
                else:
                    response = "🤖 我没听懂，输入 `help` 查看命令清单"

                # 6. 回帖
                await im_helpers.post_channel(channel_id, response)

                # 7. audit success
                await self._audit_outcome(
                    ctx, cmd_name, cmd_args, "success", None,
                    int((time.monotonic() - t0) * 1000), routed_via,
                    intent_result.confidence if intent_result else None,
                )

            except BotError as e:
                # 业务异常 — 友好提示
                await im_helpers.post_channel(channel_id, e.user_message)
                outcome = type(e).__name__.replace("Bot", "").lower().replace("error", "_error") or "handler_error"
                # 映射到 outcome enum
                if isinstance(e, BotPermissionError):
                    outcome = "permission_denied"
                elif isinstance(e, BotArgError):
                    outcome = "parse_failed"
                elif isinstance(e, BotRateLimitError):
                    outcome = "rate_limited"
                else:
                    outcome = "handler_error"
                await self._audit_outcome(
                    ctx_pre, None, None, outcome, str(e)[:1024],
                    int((time.monotonic() - t0) * 1000), "whitelist", None,
                )
            except Exception as e:
                # Pitfall 4 + N-IM-04: 不暴露 stack
                log.exception(
                    "bot.dispatch.failed",
                    extra={
                        "bot_name": self.bot_config.name,
                        "workspace_id": str(workspace_id),
                        "sender_name": sender_name,
                        "error_type": type(e).__name__,
                    },
                )
                audit_id = await self._audit_outcome(
                    ctx_pre, None, None, "handler_error",
                    f"{type(e).__name__}: {str(e)[:512]}",
                    int((time.monotonic() - t0) * 1000), "whitelist", None,
                )
                await im_helpers.post_channel(
                    channel_id,
                    f"❌ 内部错误（事件 ID: {audit_id}），请联系管理员",
                )

        async def _align_identity(
            self, sender_name: str, ctx_in: BotContext
        ) -> BotContext | None:
            """sender_name → users 表查 + workspace_id 边界（R-IM-06 + Pitfall 6）。"""
            async with self._session_factory() as session:
                stmt = select(User).where(
                    User.username == sender_name,
                    User.workspace_id == ctx_in.workspace_id,  # CLAUDE.md §2.4
                )
                result = await session.execute(stmt)
                user = result.scalar_one_or_none()

            if user is not None:
                return replace(
                    ctx_in,
                    user_id=str(user.id),
                    user_role=getattr(user, "role", None),
                    user_email=getattr(user, "email", None),
                )

            cfg = self.bot_config.identity
            if cfg.on_unknown == "reject_friendly":
                await ctx_in.im_helpers.post_channel(
                    ctx_in.channel_id,
                    cfg.unknown_hint or f"⚠️ 未识别账号 `{sender_name}`，请联系管理员",
                )
                return None
            if cfg.on_unknown == "guest_mode":
                return replace(ctx_in, user_role="guest", user_id=None)
            raise BotIdentityError("on_unknown=auto_create 留 Phase 5.E")

        async def _route_command(
            self, message: str, ctx: BotContext
        ) -> tuple[str, BotCommand | None, IntentResult | None]:
            """三层硬性顺序: keywords self_apply > parse_command > LLM intent router > ai_qa。

            Returns:
                (routed_via, cmd, intent_result) — 三个返回值二选一非空（决定 dispatch 路径）。
            """
            # Layer 1: parse_command（含 self_apply 短路 R-IM-07）
            try:
                cmd = parse_command(message, self.bot_config)
            except BotArgError:
                raise   # 让顶层异常 wrap 接

            if cmd is not None:
                routed_via = "keyword_self_apply" if cmd.self_apply else "whitelist"
                return routed_via, cmd, None

            # Layer 2: LLM intent router（Pitfall 11 — 仅在 parse 失败时调）
            cfg = self.bot_config.fallback.llm_intent_router if self.bot_config.fallback else None
            if cfg and cfg.enabled:
                result = await self.intent_router.classify(message=message, ctx=ctx)
                if (
                    result.intent != "ai_qa"
                    and result.confidence >= cfg.confidence_threshold
                ):
                    return "llm_intent", None, result
                # confidence 不足 / intent=ai_qa → 走兜底
                return "ai_qa", None, result

            # 完全无 fallback — 默认 ai_qa
            return "ai_qa", None, None

        async def _audit_outcome(
            self,
            ctx: BotContext,
            cmd_name: str | None,
            cmd_args: dict | None,
            outcome: str,
            error_message: str | None,
            latency: float | int,
            routed_via: str,
            llm_confidence: float | None,
        ) -> int | None:
            """audit log 包装 — try/except 防 audit 失败阻塞 dispatch。"""
            try:
                latency_ms = int(latency * 1000) if isinstance(latency, float) else latency
                return await self.audit.log_dispatch(
                    workspace_id=UUID(ctx.workspace_id),
                    bot_name=self.bot_config.name,
                    sender_name=ctx.user_name,
                    sender_user_id=UUID(ctx.user_id) if ctx.user_id else None,
                    cmd_name=cmd_name,
                    cmd_args=cmd_args,
                    raw_message=ctx.raw_message,
                    outcome=outcome,
                    error_message=error_message,
                    latency_ms=latency_ms,
                    routed_via=routed_via,
                    llm_confidence=llm_confidence,
                )
            except Exception:
                log.exception("bot.audit.write_failed", extra={"bot_name": self.bot_config.name})
                return None


    __all__ = ["BotDispatcher"]
    ```
