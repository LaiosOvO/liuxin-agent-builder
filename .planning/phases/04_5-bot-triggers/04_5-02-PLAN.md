```python
def load_bot_config(
    source: Path | str | dict,
    *,
    base_dir: Path | None = None,
) -> BotConfig:
    """加载 + 校验 bot.yaml。启动期 fail-fast。

    Args:
        source: Path / str 时按文件读取；dict 时直接 model_validate
        base_dir: 文件路径基准（默认 cwd 下 plugins/bots/）；source 为绝对路径时 raise

    Returns:
        BotConfig — 已通过 Pydantic v2 验证 + handler ref try-import + ${global.*} interpolation

    Raises:
        BotConfigError: 任何校验失败（明文凭据 / intents 冲突 / handler import 失败 / yaml 语法错 等）
        FileNotFoundError: yaml 文件不存在
    """


def validate_handlers(bot_config: BotConfig) -> None:
    """try-import 所有 commands.handler ref —— 失败 raise BotConfigError。

    用途：startup 时统一调用；CLI validate 子命令也调。
    """


def resolve_global_placeholders(data: dict, *, defaults: dict[str, str] | None = None) -> dict:
    """处理 ${global.<key>} 占位符 — 默认 {'default_llm': 'glm'}。

    实现：deepcopy + recursive str 替换；不引入 Jinja2（避免 dependency creep）。
    """
```

       ```python
       from __future__ import annotations
       import importlib
       import logging
       from copy import deepcopy
       from pathlib import Path
       from typing import Any

       import yaml
       from pydantic import ValidationError

       from .schemas.bot_config import BotConfig

       log = logging.getLogger(__name__)


       _DEFAULT_GLOBALS = {"default_llm": "glm"}
       _ALLOWED_BOT_DIR = "plugins/bots"


       class BotConfigError(Exception):
           pass


       def resolve_global_placeholders(
           data: dict, *, defaults: dict[str, str] | None = None
       ) -> dict:
           globals_ = {**_DEFAULT_GLOBALS, **(defaults or {})}
           # 用 deepcopy + recursive replace ${global.<key>}
           ...


       def validate_handlers(bot_config: BotConfig) -> None:
           for cmd in bot_config.commands:
               try:
                   module_path, fn_name = cmd.handler.split(":")
                   mod = importlib.import_module(module_path)
                   fn = getattr(mod, fn_name)
                   if not callable(fn):
                       raise BotConfigError(
                           f"handler {cmd.handler} 不可调用"
                       )
               except (ImportError, AttributeError, ValueError) as exc:
                   raise BotConfigError(
                       f"handler {cmd.handler} 解析失败: {exc}"
                   ) from exc


       def load_bot_config(
           source: Path | str | dict,
           *,
           base_dir: Path | None = None,
       ) -> BotConfig:
           if isinstance(source, dict):
               data = source
           else:
               yaml_path = Path(source)
               if yaml_path.is_absolute():
                   raise BotConfigError(
                       f"bot.yaml 必须是相对路径（plugins/bots/*.yaml），实际: {yaml_path}"
                   )
               if base_dir is None:
                   base_dir = Path.cwd()
               full = base_dir / yaml_path
               if _ALLOWED_BOT_DIR not in str(full):
                   raise BotConfigError(
                       f"bot.yaml 必须在 {_ALLOWED_BOT_DIR}/ 下: {full}"
                   )
               if not full.exists():
                   raise FileNotFoundError(full)
               try:
                   data = yaml.safe_load(full.read_text(encoding="utf-8"))
               except yaml.YAMLError as exc:
                   raise BotConfigError(f"yaml 语法错误 {full}: {exc}") from exc
               if not isinstance(data, dict):
                   raise BotConfigError(
                       f"yaml 根节点必须是 mapping，实际类型: {type(data).__name__}"
                   )
           # 占位符解析
           data = resolve_global_placeholders(data)
           # Pydantic 校验（触发 Pitfall 7/8 防护）
           try:
               bot_config = BotConfig.model_validate(data)
           except ValidationError as exc:
               raise BotConfigError(f"bot.yaml schema 校验失败: {exc}") from exc
           # handler import 校验
           validate_handlers(bot_config)
           log.info(
               "bot.config.loaded",
               extra={
                   "bot_name": bot_config.name,
                   "commands": [c.name for c in bot_config.commands],
                   "provider": bot_config.provider.type,
               },
           )
           return bot_config


       __all__ = [
           "BotConfigError",
           "load_bot_config",
           "validate_handlers",
           "resolve_global_placeholders",
       ]
       ```

       ```python
       import argparse
       import logging
       import sys
       from pathlib import Path

       from .loader import load_bot_config, BotConfigError


       def main(argv: list[str] | None = None) -> int:
           parser = argparse.ArgumentParser(prog="agent-builder-bot")

           validate = sub.add_parser("validate", help="校验 bot.yaml")

           args = parser.parse_args(argv)

               except (BotConfigError, FileNotFoundError) as exc:
                   print(f"✗ {args.yaml_path}: {exc}", file=sys.stderr)

           return 0
       ```
