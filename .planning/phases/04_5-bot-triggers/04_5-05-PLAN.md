       ```python
       """MattermostIMHelpers — IMHelpers Protocol 实现。

       Pitfall 9: send_dm 必须先 ensure_in_channel（MM DM channel 要求双方先建立）。
       """
       from __future__ import annotations
       import logging

       from ..context import IMHelpers   # Protocol — 仅 type hint

       log = logging.getLogger(__name__)


       class MattermostIMHelpers:
           """实现 IMHelpers Protocol — Mattermost API v4 出站。"""

           def __init__(self, *, driver, bot_user_id: str) -> None:
               """
               Args:
                   driver: AsyncDriver 实例
                   bot_user_id: 启动期 cache 的 bot user id
               """
               self._driver = driver
               self._bot_user_id = bot_user_id

           async def post_channel(self, channel_id: str, text: str) -> None:
               try:
                   await self._driver.posts.create_post(
                       options={"channel_id": channel_id, "message": text}
                   )
                   log.info(
                       "bot.listener.post_channel",
                       extra={"channel_id_prefix": channel_id[:8], "length": len(text)},
                   )
               except Exception:
                   log.exception("bot.listener.post_channel.failed")
                   raise

           async def send_dm(self, username: str, text: str) -> None:
               # Pitfall 9: 先获取目标 user_id → create_direct_channel → post_channel
               user = await self._driver.users.get_user_by_username(username=username)
               target_id = user["id"]
               dm_channel = await self._driver.channels.create_direct_channel(
                   options=[self._bot_user_id, target_id]
               )
               await self.post_channel(dm_channel["id"], text)

           async def ensure_in_channel(self, channel_id: str, user_id: str) -> None:
               try:
                   await self._driver.channels.add_user_to_channel(
                       channel_id=channel_id,
                       options={"user_id": user_id},
                   )
               except Exception:
                   # 已在 channel 内时 MM 返 400 — 容忍
                   log.debug(
                       "bot.listener.ensure_in_channel.skip",
                       extra={"channel_id_prefix": channel_id[:8], "user_id_prefix": user_id[:8]},
                   )


       __all__ = ["MattermostIMHelpers"]
       ```

    ```python
    5. structured log bot.listener.{started, message_received, reconnect_attempt} - Phase 7 钩子

    _apply_httpx_patch()

    import json
    import logging
    import random
    from collections.abc import Awaitable, Callable
    from uuid import UUID

    from mattermostautodriver import AsyncDriver

    from .mm_helpers import MattermostIMHelpers
    from ..context import IMHelpers
    # BotSettings forward reference
    from app.agent_builder.core.bot_settings import BotSettings

    log = logging.getLogger(__name__)


    _RECONNECT_BASE_DELAY = 5.0
    _RECONNECT_MAX_DELAY = 60.0
    _RECONNECT_JITTER = 2.0


    class MattermostListener:
        name: str = "mattermost"

        def __init__(
            self,
            *,
            bot_config: BotConfig,
            settings: BotSettings,
            workspace_id: UUID,
        ) -> None:
            self.bot_config = bot_config
            self._settings = settings
            self._workspace_id = workspace_id
            self._driver: AsyncDriver | None = None
            self._dispatch_fn: Callable[..., Awaitable[None]] | None = None
            self._ws_task: asyncio.Task | None = None
            self._stop_event = asyncio.Event()
            self._bot_user_id: str | None = settings.bot_user_id
            self._consecutive_failures = 0

        def register_dispatch(
            self, dispatch_fn: Callable[..., Awaitable[None]]
        ) -> None:
            self._dispatch_fn = dispatch_fn

        async def start(self) -> None:
            if self._dispatch_fn is None:
                raise RuntimeError(
                    "MattermostListener: register_dispatch 必须在 start 之前调用"
                )
            self._driver = AsyncDriver(
                {
                    "url": self._settings.url,
                    "token": self._settings.bot_token,
                    "scheme": "https" if self._settings.url.startswith("https") else "http",
                    "keepalive": True,
                    "debug": False,
                }
            )
            await self._driver.login()

            # Pitfall 1: 启动期获取 bot_user_id（cache 防 echo loop）
            if not self._bot_user_id:
                me = await self._driver.users.get_user(user_id="me")
                self._bot_user_id = me["id"]
                log.info(
                    "bot.listener.bot_user_id_cached",
                    extra={"bot_user_id_prefix": self._bot_user_id[:8]},
                )

            self._ws_task = asyncio.create_task(
                self._run_with_reconnect(), name=f"mm-listener-{self.bot_config.name}"
            )
            log.info(
                "bot.listener.started",
                extra={
                    "bot_name": self.bot_config.name,
                    "workspace_id": str(self._workspace_id),
                    "bot_username": self._settings.bot_username,
                },
            )

        async def stop(self) -> None:
            self._stop_event.set()
            if self._ws_task is not None and not self._ws_task.done():
                self._ws_task.cancel()
                try:
                    await self._ws_task
                except asyncio.CancelledError:
                    pass
            if self._driver is not None:
                try:
                    await self._driver.logout()
                except Exception:
                    log.warning("bot.listener.logout_failed", exc_info=True)
            log.info(
                "bot.listener.stopped", extra={"bot_name": self.bot_config.name}
            )

        async def _run_with_reconnect(self) -> None:
            """WS reconnect loop — Pitfall 3 指数退避 + jitter。"""
            while not self._stop_event.is_set():
                try:
                    log.info(
                        "bot.listener.ws_connecting",
                        extra={"bot_name": self.bot_config.name},
                    )
                    assert self._driver is not None
                    await self._driver.init_websocket(self._handle_event)
                    # init_websocket 正常结束意味着 WS 关闭 — reset counter
                    self._consecutive_failures = 0
                except asyncio.CancelledError:
                    raise
                except Exception as e:
                    self._consecutive_failures += 1
                    delay = min(
                        _RECONNECT_BASE_DELAY * (2 ** min(self._consecutive_failures - 1, 4)),
                        _RECONNECT_MAX_DELAY,
                    )
                    delay += random.uniform(-_RECONNECT_JITTER, _RECONNECT_JITTER)
                    delay = max(1.0, delay)
                    log.warning(
                        "bot.listener.reconnect_attempt",
                        extra={
                            "bot_name": self.bot_config.name,
                            "failure_count": self._consecutive_failures,
                            "next_delay_s": delay,
                            "error": str(e)[:200],
                        },
                    )
                    try:
                        await asyncio.wait_for(
                            self._stop_event.wait(), timeout=delay
                        )
                        return  # stop_event 已 set
                    except asyncio.TimeoutError:
                        continue

        async def _handle_event(self, event_str) -> None:
            """单条 event — 三 OR 触发 + 防 echo loop + asyncio.create_task off。"""
            try:
                event = (
                    json.loads(event_str) if isinstance(event_str, str) else event_str
                )
            except Exception:
                return
            if event.get("event") != "posted":
                return

            data = event.get("data", {})
            post_str = data.get("post")
            if not post_str:
                return
            try:
                post = (
                    json.loads(post_str) if isinstance(post_str, str) else post_str
                )
            except Exception:
                return

            # Pitfall 1: 跳过 bot 自己消息防死循环
            if post.get("user_id") == self._bot_user_id:
                return

            message = post.get("message", "").strip()
            channel_id = post.get("channel_id", "")
            channel_type = data.get("channel_type", "")
            sender_im_user_id = post.get("user_id", "")
            sender_name = data.get("sender_name", "").lstrip("@")

            # R-IM-05 三 OR 触发判定
            cfg = self.bot_config.triggers
            bot_mention = f"@{self._settings.bot_username}"
            is_dm = cfg.dm and channel_type == "D"
            is_at_mentioned = cfg.at_mention and bot_mention in message
            contains_keyword = any(kw in message for kw in cfg.keywords) if cfg.keywords else False
            if not (is_dm or is_at_mentioned or contains_keyword):
                return

            log.info(
                "bot.listener.message_received",
                extra={
                    "bot_name": self.bot_config.name,
                    "sender_name": sender_name,
                    "channel_id_prefix": channel_id[:8],
                    "is_dm": is_dm,
                    "is_at_mentioned": is_at_mentioned,
                    "contains_keyword": contains_keyword,
                },
            )

            # 剥 @<bot_username>（dispatcher 内 parser 也会剥 — 双保险）
            if is_at_mentioned:
                message = message.replace(bot_mention, "").strip()

            # Pitfall 10: asyncio.create_task off — 不阻塞 listener
            assert self._dispatch_fn is not None
            im_helpers = self._build_helpers()
            asyncio.create_task(
                self._dispatch_fn(
                    sender_name=sender_name,
                    sender_im_user_id=sender_im_user_id,
                    channel_id=channel_id,
                    channel_type=channel_type,
                    message=message,
                    workspace_id=self._workspace_id,
                    im_helpers=im_helpers,
                ),
                name=f"mm-dispatch-{self.bot_config.name}-{sender_im_user_id[:8]}",
            )

        def _build_helpers(self) -> IMHelpers:
            assert self._driver is not None and self._bot_user_id is not None
            return MattermostIMHelpers(
                driver=self._driver, bot_user_id=self._bot_user_id
            )


    __all__ = ["MattermostListener"]
    ```

    ```python
    设计要点（RESEARCH §Example 4）:
    - startup: 扫描 plugins/bots/*.yaml → load_bot_config → 构造 dispatcher + listener → start
    - shutdown: 优雅停 listeners（asyncio.gather wait_for 停止）
    - 多 bot 共存：一个 BotLifecycleManager 管多个 listener task

    注意：本 plan 提供 BotLifecycleManager 类；main.py 真实接入 lifespan 由执行者按既有 main.py 模式追加。
    """
    from __future__ import annotations
    import asyncio
    import logging
    from collections.abc import Awaitable, Callable
    from pathlib import Path
    from uuid import UUID

    from ..bot_dispatcher.audit import AuditLogger
    from ..bot_dispatcher.dispatcher import BotDispatcher
    from ..bot_dispatcher.idempotency import IdempotencyStore
    from ..bot_dispatcher.listeners.mattermost import MattermostListener
    from ..bot_dispatcher.llm_provider import get_llm_provider
    from ..bot_dispatcher.llm_router import BotIntentRouter
    from ..bot_dispatcher.loader import load_bot_config, BotConfigError
    from ..bot_dispatcher.rate_limiter import RateLimiter
    from ..bot_dispatcher.registry import HandlerRegistry
    from ..core.bot_settings import BotSettings

    log = logging.getLogger(__name__)


    class BotLifecycleManager:
        def __init__(self) -> None:
            self._listeners: dict[str, MattermostListener] = {}

        async def startup(
            self,
            *,
            session_factory: Callable[[], Awaitable],
            redis,
            bot_yaml_dir: Path | None = None,
            default_workspace_id: UUID | None = None,
        ) -> None:
            """启动所有 bot.yaml 对应的 listener。

            v1 简化：default_workspace_id 注入（多 workspace 留 v1.5 — RESEARCH Open Q #5）。
            """
            yaml_dir = bot_yaml_dir or Path("plugins/bots")
            if not yaml_dir.exists():
                log.info("bot.lifecycle.no_bots", extra={"dir": str(yaml_dir)})
                return

            yaml_files = sorted(yaml_dir.glob("*.yaml"))
            if not yaml_files:
                log.info("bot.lifecycle.no_yaml_files", extra={"dir": str(yaml_dir)})
                return

            workspace_id = default_workspace_id or UUID(int=0)  # placeholder
            audit = AuditLogger(session_factory=session_factory)
            rate_limiter_default = RateLimiter(redis=redis, per_user_per_minute=10)
            idempotency = IdempotencyStore(redis=redis)

            for yaml_path in yaml_files:
                try:
                    bot_config = load_bot_config(yaml_path)
                    settings = BotSettings()  # 从 env 读 MM_*
                    llm_provider = get_llm_provider(
                        bot_config.fallback.llm_intent_router.llm
                        if bot_config.fallback and bot_config.fallback.llm_intent_router
                        else "glm"
                    )

                    # per-bot 速率限制（如有自定义）
                    per_min = (
                        bot_config.audit.rate_limit.per_user_per_minute
                        if bot_config.audit and bot_config.audit.rate_limit
                        else 10
                    )
                    rate_limiter = RateLimiter(redis=redis, per_user_per_minute=per_min)

                    registry = HandlerRegistry(bot_config=bot_config)
                    registry.validate_all_handlers()  # 启动期 fail-fast

                    intent_router = BotIntentRouter(
                        bot_config=bot_config, llm_provider=llm_provider
                    )

                    dispatcher = BotDispatcher(
                        bot_config=bot_config,
                        registry=registry,
                        intent_router=intent_router,
                        rate_limiter=rate_limiter,
                        idempotency=idempotency,
                        audit=audit,
                        session_factory=session_factory,
                    )

                    if bot_config.provider.type == "mattermost":
                        listener = MattermostListener(
                            bot_config=bot_config,
                            settings=settings,
                            workspace_id=workspace_id,
                        )
                        listener.register_dispatch(dispatcher.dispatch_message)
                        await listener.start()
                        self._listeners[bot_config.name] = listener
                        log.info(
                            "bot.lifecycle.started",
                            extra={
                                "bot_name": bot_config.name,
                                "provider": bot_config.provider.type,
                                "commands": [c.name for c in bot_config.commands],
                                "workspace_id": str(workspace_id),
                            },
                        )
                    else:
                        log.warning(
                            "bot.lifecycle.unsupported_provider",
                            extra={
                                "bot_name": bot_config.name,
                                "provider": bot_config.provider.type,
                            },
                        )
                except (BotConfigError, FileNotFoundError) as e:
                    log.exception(
                        "bot.lifecycle.start_failed",
                        extra={"yaml_path": str(yaml_path), "error": str(e)[:200]},
                    )
                    # v1 严格：一个 bot 起不来阻断启动（防 silent 配置错误）
                    raise

        async def shutdown(self) -> None:
            for name, listener in self._listeners.items():
                try:
                    await asyncio.wait_for(listener.stop(), timeout=10.0)
                    log.info("bot.lifecycle.stopped", extra={"bot_name": name})
                except asyncio.TimeoutError:
                    log.warning(
                        "bot.lifecycle.stop_timeout",
                        extra={"bot_name": name},
                    )
                except Exception as e:
                    log.warning(
                        "bot.lifecycle.stop_failed",
                        extra={"bot_name": name, "error": str(e)[:200]},
                    )
            self._listeners.clear()


    __all__ = ["BotLifecycleManager"]
    ```
