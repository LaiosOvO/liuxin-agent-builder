       """LLMProvider Protocol — bot dispatcher LLM 调用抽象（v1 GLM 默认）。

       Phase 4.5 v1 仅 GLM；Phase 5.E 扩展 OpenAI / 自托管时此模块 factory 加新 case。

       设计：
       - Protocol + @runtime_checkable — 单元测试可 mock 实现 LLMProvider
       - complete() 接受 (prompt, *, max_tokens, timeout_seconds) 返回 str
       - timeout 由调用方 asyncio.timeout 控制（provider 内部也可设 SDK timeout 兜底）
       """
       from __future__ import annotations
       import logging
       import os
       from typing import Protocol, runtime_checkable

       log = logging.getLogger(__name__)


       @runtime_checkable
       class LLMProvider(Protocol):
           name: str

           async def complete(
               self, prompt: str, *, max_tokens: int = 500, timeout_seconds: float = 5.0
           ) -> str: ...


       class GLMProvider:
           name: str = "glm"

           def __init__(self, *, api_key: str | None = None) -> None:
               self._api_key = api_key or os.environ.get("ZHIPUAI_API_KEY")
               if not self._api_key:
                   raise ValueError(
                       "GLMProvider: 缺少 ZHIPUAI_API_KEY 环境变量"
                   )

           async def complete(
               self, prompt: str, *, max_tokens: int = 500, timeout_seconds: float = 5.0
           ) -> str:
               # 调 zhipuai SDK；timeout 通过 SDK timeout 参数 + 外层 asyncio.timeout 双保险
               # 注意：zhipuai SDK 不支持 async — 用 asyncio.to_thread 包装
               import asyncio
               from zhipuai import ZhipuAI

               client = ZhipuAI(api_key=self._api_key)

               def _sync_call() -> str:
                   resp = client.chat.completions.create(
                       model="glm-4",
                       messages=[{"role": "user", "content": prompt}],
                       max_tokens=max_tokens,
                       timeout=timeout_seconds,
                   )
                   return resp.choices[0].message.content or ""

               return await asyncio.to_thread(_sync_call)


       _PROVIDERS: dict[str, type] = {"glm": GLMProvider}


       def get_llm_provider(name: str, **kwargs) -> LLMProvider:
           cls = _PROVIDERS.get(name.lower())
           if cls is None:
               raise ValueError(
                   f"未知 LLM provider: {name}（v1 仅支持 {list(_PROVIDERS)}; Phase 5.E 扩展）"
               )
           return cls(**kwargs)


       __all__ = ["LLMProvider", "GLMProvider", "get_llm_provider"]
       ```

    ```python
    """LLM-based intent router — Phase 4.5 R-IM-04 + N-IM-05。

    设计要点：
    - asyncio.timeout (Python 3.11+) 包装 LLM 调用 — 超时立即 ai_qa 兜底
    - _strip_json_fences 剥离 ```json fence + 前缀文本（Pitfall 5）
    - parse_failed 时降级为 ai_qa + 保留 raw 输出（不丢消息）
    - structured log bot.dispatch.llm.* 命名空间（Phase 7 Run Viewer 钩子）
    """
    from __future__ import annotations
    import asyncio
    import json
    import logging
    import re
    from dataclasses import dataclass, field
    from pathlib import Path
    from typing import TYPE_CHECKING, Any

    import jinja2

    from .schemas.bot_config import BotConfig
    from .llm_provider import LLMProvider

    if TYPE_CHECKING:
        from .context import BotContext  # Wave 3 (Plan 04) 才定义

    log = logging.getLogger(__name__)


    _FENCE_RE = re.compile(r"```(?:json)?\s*(.+?)\s*```", re.S)
    _PROMPTS_DIR = Path(__file__).parent / "prompts"


    @dataclass(frozen=True)
    class IntentResult:
        intent: str
        args: dict[str, Any]
        confidence: float
        ai_reply: str | None = None
        raw: str = ""


    def _strip_json_fences(text: str) -> str:
        text = text.strip()
        m = _FENCE_RE.search(text)
        return m.group(1).strip() if m else text


    class BotIntentRouter:
        def __init__(self, bot_config: BotConfig, llm_provider: LLMProvider) -> None:
            self.bot_config = bot_config
            self.llm = llm_provider
            self._tpl_env = jinja2.Environment(
                loader=jinja2.FileSystemLoader(searchpath=str(_PROMPTS_DIR)),
                autoescape=False,
                keep_trailing_newline=True,
            )

        async def classify(
            self, *, message: str, ctx: "BotContext"
        ) -> IntentResult:
            cfg = self.bot_config.fallback.llm_intent_router if self.bot_config.fallback else None
            if not (cfg and cfg.enabled):
                return IntentResult(intent="ai_qa", args={}, confidence=0.0)

            tpl = self._tpl_env.get_template(cfg.prompt_template_path)
            prompt = tpl.render(
                bot_name=self.bot_config.name,
                intents=cfg.intents,
                user_message=message,
                sender_username=ctx.user_name,
                sender_role=ctx.user_role or "未注册",
            )

            log.info(
                "bot.dispatch.llm.attempt",
                extra={
                    "bot_name": self.bot_config.name,
                    "workspace_id": ctx.workspace_id,
                    "user_name": ctx.user_name,
                    "timeout_seconds": cfg.timeout_seconds,
                },
            )

            try:
                async with asyncio.timeout(cfg.timeout_seconds):
                    resp = await self.llm.complete(
                        prompt,
                        max_tokens=600,  # 比 ai_qa 大；够 JSON + ai_reply
                        timeout_seconds=cfg.timeout_seconds,
                    )
            except (asyncio.TimeoutError, Exception) as e:
                log.warning(
                    "bot.dispatch.llm.timeout",
                    extra={
                        "bot_name": self.bot_config.name,
                        "workspace_id": ctx.workspace_id,
                        "error": type(e).__name__,
                        "msg": str(e)[:200],
                    },
                )
                return IntentResult(intent="ai_qa", args={}, confidence=0.0)

            stripped = _strip_json_fences(resp)
            try:
                data = json.loads(stripped)
            except json.JSONDecodeError:
                log.warning(
                    "bot.dispatch.llm.parse_failed",
                    extra={
                        "bot_name": self.bot_config.name,
                        "raw": resp[:200],
                    },
                )
                return IntentResult(
                    intent="ai_qa",
                    args={},
                    confidence=0.0,
                    ai_reply=resp.strip()[:1000],
                    raw=resp,
                )

            return IntentResult(
                intent=str(data.get("intent", "ai_qa")).strip().lower(),
                args=data.get("args", {}) or {},
                confidence=float(data.get("confidence", 0.0)),
                ai_reply=(data.get("ai_reply") or "").strip() or None,
                raw=resp,
            )

        async def answer_ai_qa(
            self, *, message: str, ctx: "BotContext"
        ) -> str:
            ai_qa_cfg = self.bot_config.fallback.ai_qa if self.bot_config.fallback else None
            if not (ai_qa_cfg and ai_qa_cfg.enabled):
                return "🤖 我没听懂，输入 `help` 查看命令清单"

            tpl_path = ai_qa_cfg.prompt_template_path or "ai_qa_zh.md"
            tpl = self._tpl_env.get_template(tpl_path)
            prompt = tpl.render(
                bot_name=self.bot_config.name,
                bot_description=self.bot_config.description,
                system_message=ai_qa_cfg.system_message,
                user_message=message,
            )

            try:
                async with asyncio.timeout(ai_qa_cfg.max_tokens / 100 + 5):
                    # 简单 heuristic: 500 tokens ≈ 10s 上限
                    resp = await self.llm.complete(
                        prompt,
                        max_tokens=ai_qa_cfg.max_tokens,
                        timeout_seconds=10.0,
                    )
                return resp.strip()
            except (asyncio.TimeoutError, Exception) as e:
                log.warning(
                    "bot.dispatch.ai_qa.timeout",
                    extra={"bot_name": self.bot_config.name, "error": str(e)[:200]},
                )
                return "⏳ AI 处理超时，请稍后再试或输入 `help` 查看命令"


    __all__ = ["BotIntentRouter", "IntentResult", "_strip_json_fences"]
    ```
