       ```python
       """httpx 0.28+ vs mattermostautodriver 2.0 兼容垫片（Pitfall 2 P0）。

       httpx 0.28 移除 `proxies` kwarg，mattermostautodriver 2.0 仍传 → TypeError。
       必须在 `import mattermostautodriver` 之前调用 `apply_httpx_patch()`。

       参考: hr/offboarding-flow workers/mattermost_listener.py:30-40 生产验证 1.5 个月。
       """
       from __future__ import annotations
       import logging
       import httpx

       log = logging.getLogger(__name__)
       _PATCHED = False
       _ORIG_INIT = None


       def apply_httpx_patch() -> None:
           """幂等 — 多次调用仅生效一次。"""
           global _PATCHED, _ORIG_INIT
           if _PATCHED:
               return
           _ORIG_INIT = httpx.AsyncClient.__init__

           def _patched_init(self, *args, **kwargs):
               kwargs.pop("proxies", None)
               return _ORIG_INIT(self, *args, **kwargs)

           httpx.AsyncClient.__init__ = _patched_init  # type: ignore[method-assign]
           _PATCHED = True
           log.info("bot.httpx_patch.applied")


       def is_patched() -> bool:
           return _PATCHED


       __all__ = ["apply_httpx_patch", "is_patched"]
       ```
