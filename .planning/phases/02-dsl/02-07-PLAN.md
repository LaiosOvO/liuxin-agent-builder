**stream_mode="updates"**：LangGraph 1.2 此模式在每个节点完成后回吐一个 {node_id: state_delta} dict；不需要 deltas 模式。

# backend/app/agent_builder/workflow/event_bus.py
import json
from datetime import datetime
from uuid import UUID

from redis.asyncio import Redis

EVENT_LOG_MAX = 3600  # 保留约 1 小时
EVENT_LOG_TTL = 3600  # Redis Stream key TTL

def _pubsub_channel(instance_id: UUID) -> str:
    return f"agent_builder:instance_events:{instance_id}"

def _stream_key(instance_id: UUID) -> str:
    return f"agent_builder:instance_event_log:{instance_id}"

def _seq_key(instance_id: UUID) -> str:
    return f"agent_builder:instance_event_seq:{instance_id}"


class EventBus:
    def __init__(self, redis: Redis):
        self.redis = redis
    
    async def publish(self, instance_id: UUID, event: str, data: dict) -> dict:
        seq = await self.redis.incr(_seq_key(instance_id))
        packet = {
            "id": seq,
            "event": event,
            "instance_id": str(instance_id),
            "timestamp": datetime.utcnow().isoformat() + "Z",
            "data": data,
        }
        encoded = json.dumps(packet, ensure_ascii=False)
        # XADD + PUBLISH
        await self.redis.xadd(_stream_key(instance_id), {"packet": encoded}, maxlen=EVENT_LOG_MAX, approximate=True)
        await self.redis.expire(_stream_key(instance_id), EVENT_LOG_TTL)
        await self.redis.publish(_pubsub_channel(instance_id), encoded)
        return packet
    
    async def subscribe(self, instance_id: UUID):
        """async generator: yield events from pub/sub forever."""
        pubsub = self.redis.pubsub()
        await pubsub.subscribe(_pubsub_channel(instance_id))
        try:
            async for msg in pubsub.listen():
                if msg.get("type") == "message":
                    yield json.loads(msg["data"])
        finally:
            await pubsub.unsubscribe(_pubsub_channel(instance_id))
            await pubsub.close()
    
    async def replay_from_seq(self, instance_id: UUID, last_seq: int):
        """从 Redis Stream 取 seq > last_seq 的事件回放。"""
        entries = await self.redis.xrange(_stream_key(instance_id), "-", "+")
        for entry_id, fields in entries:
            packet = json.loads(fields["packet"])
            if packet["id"] > last_seq:
                yield packet
```

# backend/app/agent_builder/api/v1/instances_events.py
from fastapi import APIRouter, Depends, Header, HTTPException, Request
from sse_starlette.sse import EventSourceResponse, ServerSentEvent
from uuid import UUID

router = APIRouter()

@router.get("/instances/{instance_id}/events")
async def stream_events(
    instance_id: UUID,
    request: Request,
    user = Depends(get_current_user),
    last_event_id: str | None = Header(default=None, alias="Last-Event-ID"),
):
    instance = await load_instance(instance_id)
    if not instance or instance.workspace_id != user.current_workspace_id:
        raise HTTPException(404)
    
    redis = get_redis()
    bus = EventBus(redis)
    
    async def event_generator():
        # 1. 补发 Last-Event-ID 之后的事件
        last_seq = int(last_event_id) if last_event_id else 0
        async for packet in bus.replay_from_seq(instance_id, last_seq):
            if await request.is_disconnected(): return
            yield ServerSentEvent(id=str(packet["id"]), event=packet["event"], data=json.dumps(packet))
        # 2. 订阅实时事件
        async for packet in bus.subscribe(instance_id):
            if await request.is_disconnected(): return
            yield ServerSentEvent(id=str(packet["id"]), event=packet["event"], data=json.dumps(packet))
            if packet["event"] in ("instance.complete", "instance.failed"):
                return  # 实例完成自然结束 SSE
    
    return EventSourceResponse(event_generator(), media_type="text/event-stream")

# backend/app/agent_builder/workflow/runner.py
async def run_instance(instance_id: UUID):
    """实际跑实例的核心逻辑。
    
    入口：
    - arq worker（生产环境）
    - asyncio.create_task（开发/测试 fast path）
    """
    redis = get_redis()
    bus = EventBus(redis)
    
    async with get_session() as db:
        instance = await load_instance(db, instance_id)
        await mark_instance_running(db, instance_id)
    
    try:
        async with get_checkpointer() as checkpointer:
            compiler = DSLCompiler(
                workspace_id=instance.workspace_id,
                instance_id=instance.id,
                redis=redis,
                event_bus=bus,
            )
            graph = compiler.compile(instance.dsl_snapshot, checkpointer)
            thread_id = build_thread_id(instance.workspace_id, instance.id)
            cfg = {"configurable": {"thread_id": thread_id}}
            
            initial_input = instance.initial_input or {}
            async for chunk in graph.astream(initial_input, cfg, stream_mode="updates"):
                # chunk = {node_id: state_delta_dict}
                for node_id, _delta in chunk.items():
                    async with get_session() as db:
                        await upsert_node_state(db, instance_id, node_id, status="completed")
                    await bus.publish(instance_id, EVENT_NODE_COMPLETE, {"node_id": node_id})
            
            async with get_session() as db:
                await mark_instance_completed(db, instance_id)
            await bus.publish(instance_id, EVENT_INSTANCE_COMPLETE, {"final_status": "completed"})
    
    except NodeExecutionError as e:
        async with get_session() as db:
            await upsert_node_state(db, instance_id, e.node_id, status="failed", error_message=str(e))
            await mark_instance_failed(db, instance_id, str(e))
        await bus.publish(instance_id, EVENT_NODE_ERROR, {"node_id": e.node_id, "error_message": str(e)})
        await bus.publish(instance_id, EVENT_INSTANCE_FAILED, {"error_message": str(e), "failed_node_id": e.node_id})
    
    except Exception as e:
        # 兜底失败
        async with get_session() as db:
            await mark_instance_failed(db, instance_id, f"unexpected: {e}")
        await bus.publish(instance_id, EVENT_INSTANCE_FAILED, {"error_message": str(e), "failed_node_id": None})
```
