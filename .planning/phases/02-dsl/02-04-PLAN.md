```python
# backend/app/agent_builder/workflow/nodes/base.py
from abc import ABC, abstractmethod
from typing import Any
import asyncio

from jinja2.sandbox import SandboxedEnvironment
from tenacity import AsyncRetrying, RetryError, stop_after_attempt, wait_exponential, retry_if_exception_type

from app.agent_builder.workflow.jinja_env import build_jinja_env

class NodeExecutionError(Exception):
    """节点执行失败（重试耗尽 / 不可重试错误）"""
    def __init__(self, node_id: str, message: str, original: Exception | None = None):
        self.node_id = node_id
        self.message = message
        self.original = original
        super().__init__(f"[{node_id}] {message}")

class BaseNodeExecutor(ABC):
    def __init__(self, node_def: dict, *, jinja_env: SandboxedEnvironment | None = None):
        self.node_def = node_def
        self.node_id = node_def["id"]
        self.node_type = node_def["type"]
        self.config = node_def.get("config", {})
        self.jinja_env = jinja_env or build_jinja_env()
    
    async def __call__(self, state: dict) -> dict:
        """LangGraph node fn 入口。state 是当前 state dict（含 pointer 已透明解析）。"""
        rendered_config = self._render_config(state)
        retry_count = rendered_config.get("retry_count", self._default_retry_count())
        backoff = rendered_config.get("backoff_base_sec", 1)
        timeout = rendered_config.get("timeout_sec", self._default_timeout())
        
        async def _run():
            return await asyncio.wait_for(
                self.execute(rendered_config, state),
                timeout=timeout,
            )
        
        try:
            async for attempt in AsyncRetrying(
                stop=stop_after_attempt(retry_count + 1),
                wait=wait_exponential(multiplier=backoff, max=60),
                retry=retry_if_exception_type(self._retryable_exceptions()),
                reraise=True,
            ):
                with attempt:
                    result = await _run()
        except (asyncio.TimeoutError, *self._retryable_exceptions()) as e:
            raise NodeExecutionError(self.node_id, f"重试 {retry_count + 1} 次仍失败: {e!s}", original=e)
        except Exception as e:
            raise NodeExecutionError(self.node_id, str(e), original=e)
        
        return {self.node_id: result}  # LangGraph 自动 merge
    
    @abstractmethod
    async def execute(self, config: dict, state: dict) -> dict | Any:
        """子类实现：返回值会以 state[self.node_id] 形式写入 state"""
        ...
    
    def _render_config(self, state: dict) -> dict:
        """递归用 Jinja2 渲染 config 中所有 str 字段（含嵌套 dict / list）"""
        def render(value):
            if isinstance(value, str):
                return self.jinja_env.from_string(value).render(state)
            if isinstance(value, dict):
                return {k: render(v) for k, v in value.items()}
            if isinstance(value, list):
                return [render(v) for v in value]
            return value
        return {k: render(v) for k, v in self.config.items()}
    
    def _default_retry_count(self) -> int: return 0
    def _default_timeout(self) -> int: return 30
    def _retryable_exceptions(self) -> tuple[type[Exception], ...]: return ()
```

**为何放在 BaseNodeExecutor 而非 LangGraph 框架**：
- LangGraph 1.2 节点函数无内置重试
- tenacity 是 Python 异步重试标准库，已经 in deps
- 把渲染、重试、超时、事件发布集中在基类，子类只写真实业务逻辑

```python
# backend/app/agent_builder/workflow/tool_registry.py
from typing import Callable, Awaitable, Any

TOOL_REGISTRY: dict[str, Callable[..., Awaitable[Any]]] = {}

def register_tool(name: str):
    """Decorator: 注册一个 async function 为 tool。
    
    Example:
        @register_tool("echo")
        async def echo(*, text: str) -> dict:
            return {"echoed": text}
    """
    if not name.replace("_", "").replace(".", "").isalnum():
        raise ValueError(f"非法 tool 名: {name}")
    def deco(fn):
        if name in TOOL_REGISTRY:
            raise ValueError(f"Tool {name} 已注册")
        TOOL_REGISTRY[name] = fn
        return fn
    return deco

def get_registered_tool(name: str) -> Callable | None:
    return TOOL_REGISTRY.get(name)

# 内置 demo 工具（用于 E2E）
@register_tool("echo")
async def echo(*, text: str) -> dict:
    return {"echoed": text, "length": len(text)}

@register_tool("uppercase")
async def uppercase(*, text: str) -> dict:
    return {"result": text.upper()}
```

```python
# backend/app/agent_builder/workflow/nodes/if_else.py
from app.agent_builder.workflow.nodes.base import BaseNodeExecutor

class IfElseNodeExecutor(BaseNodeExecutor):
    async def execute(self, config: dict, state: dict) -> dict:
        """if_else 不写 state（仅用作路由），返回 _routed_to: target_id 仅供事件用。"""
        target = self.resolve_route(config, state)
        return {"_routed_to": target}
    
    def resolve_route(self, config: dict, state: dict) -> str:
        """对 conditions[].expr 依次 Jinja 求值（布尔），匹配则返回 target_node_id；否则 default_target。"""
        for cond in config["conditions"]:
            expr_str = cond["expr"]
            try:
                template = self.jinja_env.from_string(expr_str)
                rendered = template.render(state).strip().lower()
                if rendered in ("true", "1", "yes"):
                    return cond["target_node_id"]
            except Exception:
                continue  # expr 求值失败，跳过该 condition
        return config["default_target"]
```

```python
# backend/app/agent_builder/workflow/nodes/tool.py
import httpx
from app.agent_builder.workflow.nodes.base import BaseNodeExecutor
from app.agent_builder.workflow.tool_registry import get_registered_tool

class ToolNodeExecutor(BaseNodeExecutor):
    def _default_retry_count(self) -> int: return 2
    def _default_timeout(self) -> int: return 10
    def _retryable_exceptions(self):
        return (httpx.RequestError, httpx.TimeoutException, httpx.HTTPStatusError)
    
    async def execute(self, config: dict, state: dict) -> dict:
        kind = config["kind"]
        if kind == "http":
            return await self._exec_http(config, state)
        if kind == "python":
            return await self._exec_python(config, state)
        raise ValueError(f"Unknown tool kind: {kind}")
    
    async def _exec_http(self, config: dict, state: dict) -> dict:
        async with httpx.AsyncClient(timeout=config.get("timeout_sec", 10)) as client:
            response = await client.request(
                method=config["method"],
                url=config["url"],
                headers=config.get("headers", {}),
                json=config.get("body") if isinstance(config.get("body"), (dict, list)) else None,
                data=config.get("body") if isinstance(config.get("body"), str) else None,
            )
            response.raise_for_status()
            try:
                body = response.json()
            except ValueError:
                body = response.text
            return {
                "output": body,
                "status_code": response.status_code,
                "headers": dict(response.headers),
            }
    
    async def _exec_python(self, config: dict, state: dict) -> dict:
        fn_name = config["function"]
        fn = get_registered_tool(fn_name)
        if not fn:
            raise NodeExecutionError(self.node_id, f"Tool '{fn_name}' 未注册")
        args = config.get("args", {})
        result = await fn(**args)
        return {"result": result}
```
