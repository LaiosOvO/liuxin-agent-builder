# backend/app/agent_builder/services/workflow_service.py
class WorkflowService:
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def create_workflow(self, *, workspace_id: UUID, name: str, description: str | None, user_id: UUID) -> Workflow:
        # 创建 Workflow + 初始 draft version（空 DSL，仅 start + end）
        wf = Workflow(workspace_id=workspace_id, name=name, description=description, created_by=user_id, status="draft")
        self.db.add(wf)
        await self.db.flush()
        initial_dsl = {
            "version": "1.0", "name": name, "state_schema": {},
            "nodes": [
                {"id": "start", "type": "start", "position": {"x": 100, "y": 200}, "config": {}},
                {"id": "end", "type": "end", "position": {"x": 500, "y": 200}, "config": {}},
            ],
            "edges": [{"id": "e1", "source": "start", "target": "end"}],
        }
        wv = WorkflowVersion(workspace_id=workspace_id, workflow_id=wf.id, version_no=0, kind="draft", dsl=initial_dsl, created_by=user_id)
        self.db.add(wv); await self.db.flush()
        wf.current_draft_version_id = wv.id
        await self.db.commit()
        return wf
    
    async def save_draft(self, *, workflow_id: UUID, workspace_id: UUID, dsl: dict, user_id: UUID) -> tuple[Workflow, list[ValidationError]]:
        # 校验（warning 可放行）
        errors = DSLValidator().validate(dsl)
        fatal = [e for e in errors if e.severity == "error"]
        if fatal:
            return None, errors  # 上层返 422
        
        # 更新 draft version
        wf = await self._load_workflow(workflow_id, workspace_id)
        existing_draft = await self._load_version(wf.current_draft_version_id, workspace_id) if wf.current_draft_version_id else None
        if existing_draft:
            existing_draft.dsl = dsl
            existing_draft.created_by = user_id
        else:
            wv = WorkflowVersion(workspace_id=workspace_id, workflow_id=workflow_id, version_no=0, kind="draft", dsl=dsl, created_by=user_id)
            self.db.add(wv); await self.db.flush()
            wf.current_draft_version_id = wv.id
        await self.db.commit()
        return wf, errors  # errors 中可能含 warning，作信息返回
    
    async def publish(self, *, workflow_id: UUID, workspace_id: UUID, user_id: UUID) -> Workflow:
        wf = await self._load_workflow(workflow_id, workspace_id)
        draft = await self._load_version(wf.current_draft_version_id, workspace_id)
        errors = DSLValidator().validate(draft.dsl)
        fatal = [e for e in errors if e.severity == "error"]
        if fatal:
            raise DSLCompilationError(fatal)  # 422
        
        # 新建发布 version
        prev_no = await self._max_published_version_no(workflow_id, workspace_id)
        wv = WorkflowVersion(workspace_id=workspace_id, workflow_id=workflow_id, version_no=prev_no+1, kind="published", dsl=draft.dsl, created_by=user_id)
        self.db.add(wv); await self.db.flush()
        wf.current_published_version_id = wv.id
        wf.status = "published"
        await self.db.commit()
        return wf
```

# backend/app/agent_builder/services/instance_service.py
class InstanceService:
    def __init__(self, db: AsyncSession, engine: ExecutionEngine):
        self.db = db
        self.engine = engine
    
    async def start(self, *, workflow_id: UUID, workspace_id: UUID, initial_input: dict, user_id: UUID) -> FlowInstance:
        wf = await load_workflow(self.db, workflow_id, workspace_id)
        if not wf.current_published_version_id:
            raise HTTPException(409, "工作流尚未发布")
        return await self.engine.start_instance(
            workflow_version_id=wf.current_published_version_id,
            initial_input=initial_input, user_id=user_id, workspace_id=workspace_id,
        )
    
    async def list(self, *, workspace_id: UUID, workflow_id: UUID | None = None, status: str | None = None,
                   search: str | None = None, page: int = 1, page_size: int = 20) -> tuple[list[FlowInstance], int]:
        stmt = WorkspaceScopedQuery.select(FlowInstance)
        if workflow_id:
            stmt = stmt.where(FlowInstance.workflow_id == workflow_id)
        if status:
            stmt = stmt.where(FlowInstance.status == status)
        if search:
            stmt = stmt.where(FlowInstance.id.cast(String).ilike(f"%{search}%"))
        total = await self.db.scalar(select(func.count()).select_from(stmt.subquery()))
        stmt = stmt.order_by(FlowInstance.created_at.desc()).offset((page-1)*page_size).limit(page_size)
        rows = (await self.db.execute(stmt)).scalars().all()
        return list(rows), total
    
    async def get(self, *, instance_id: UUID, workspace_id: UUID) -> InstanceDetail:
        stmt = WorkspaceScopedQuery.select(FlowInstance).where(FlowInstance.id == instance_id)
        inst = (await self.db.execute(stmt)).scalar_one_or_none()
        if not inst:
            raise HTTPException(404)
        node_stmt = WorkspaceScopedQuery.select(NodeState).where(NodeState.instance_id == instance_id).order_by(NodeState.started_at)
        node_states = (await self.db.execute(node_stmt)).scalars().all()
        return InstanceDetail(instance=inst, node_states=list(node_states))
    
    async def abort(self, *, instance_id: UUID, workspace_id: UUID, user_id: UUID) -> FlowInstance:
        return await self.engine.abort_instance(instance_id, user_id=user_id, workspace_id=workspace_id)
```

"use client";
import { useEffect } from "react";

export function SseListener({ instanceId, onEvent }: { instanceId: string; onEvent: (e: any) => void }) {
  useEffect(() => {
    const source = new EventSource(
      `/api/agent_builder/v1/instances/${instanceId}/events`,
      { withCredentials: true },
    );
    source.onmessage = (ev) => onEvent(JSON.parse(ev.data));
    ["node.start", "node.complete", "node.error", "instance.complete", "instance.failed"].forEach(t => {
      source.addEventListener(t, (ev: any) => onEvent({ event: t, ...JSON.parse(ev.data) }));
    });
    source.onerror = () => { /* EventSource auto-reconnect; do nothing */ };
    return () => source.close();
  }, [instanceId, onEvent]);
  return null;
}
