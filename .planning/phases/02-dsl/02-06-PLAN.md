<pointer_design>
**Pointer 格式**：`__ptr__:redis:state:<32位 hex uuid>`
- 前缀 `__ptr__:redis:state:` 用于扫描识别（极少冲突）
- uuid 用 `uuid.uuid4().hex`（32 位 hex，不会与正常数据冲突）

**Redis key 命名**：`agent_builder:state_ptr:<workspace_id>:<instance_id>:<uuid>`
- workspace_id 前缀防租户碰撞
- instance_id 便于实例完成后批量清理（Phase 7 加 TTL 自动清理）

**TTL**：30 天（CONTEXT.md 锁定）

**判定阈值**：单字段 JSON 编码后 UTF-8 字节数 > 4096

**算法**：
- 写入：递归扫描 state_delta dict 的**顶层**字段；每个 value 计算 JSON size，超阈值则存 Redis 替换为 pointer
- 读取：递归扫描 state（所有 dict / list 嵌套），遇到字符串值匹配 pointer 格式时从 Redis 拉回

**为什么写入只看顶层，读取看全部嵌套**：
- 写入时：节点输出整体被 `{node_id: <output>}` 包裹，顶层字段就是 node_id；output 是一个 dict（如 `{message: <text>, usage: {...}}`），我们对它的二级字段挑大值替换
- 读取时：state 已经被多个节点写入，pointer 可能嵌套很深；read 必须递归处理

**实际实现的「写入第二层」**：BaseNodeExecutor 收到 execute() 返回的 dict（如 `{message: "...10KB...", role: "assistant", usage: {...}}`），扫描这个 dict 的每个 value：
```python
# 节点返回 {message: "10KB string", role: "assistant", usage: {...}}
# write_state_with_pointers 后变成
# {message: "__ptr__:redis:state:abc...", role: "assistant", usage: {...}}
# 然后 BaseNodeExecutor 返回 {self.node_id: <swapped>}
# 整体进入 state
```
</pointer_design>

# backend/app/agent_builder/workflow/state_pointer.py
import json
import uuid
from typing import Any
from uuid import UUID

from redis.asyncio import Redis

LARGE_THRESHOLD_BYTES = 4096
POINTER_PREFIX = "__ptr__:redis:state:"
REDIS_KEY_PREFIX = "agent_builder:state_ptr"
TTL_SECONDS = 30 * 86400  # 30 天


def is_pointer(value: Any) -> bool:
    return isinstance(value, str) and value.startswith(POINTER_PREFIX)


def parse_pointer(pointer: str) -> str:
    """从 pointer 字符串提取 uuid 部分。"""
    if not is_pointer(pointer):
        raise ValueError(f"非法 pointer: {pointer}")
    return pointer[len(POINTER_PREFIX):]


def _redis_key(workspace_id: UUID, instance_id: UUID, ptr_uuid: str) -> str:
    return f"{REDIS_KEY_PREFIX}:{workspace_id}:{instance_id}:{ptr_uuid}"


async def write_state_with_pointers(
    state_delta: dict,
    *,
    workspace_id: UUID,
    instance_id: UUID,
    redis: Redis,
) -> dict:
    """对 state_delta 中每个字段判断大小，大于阈值则写 Redis 替换为 pointer。
    
    返回新 dict（不修改原 dict，遵循 immutability 规则）。
    
    Args:
        state_delta: 节点 execute() 返回的输出字典（如 LLM 节点 {message, role, usage, model}）
        workspace_id: 用于 Redis key 命名空间
        instance_id: 同上
        redis: Redis async 客户端
    
    Returns:
        新字典，大字段已被 pointer 替换
    """
    if not isinstance(state_delta, dict):
        return state_delta  # 节点返回不是 dict（如 if_else），不处理
    
    out = {}
    for key, value in state_delta.items():
        try:
            encoded = json.dumps(value, ensure_ascii=False, default=str)
        except (TypeError, ValueError):
            out[key] = value  # 不可序列化，保留原值（让 LangGraph 自己处理）
            continue
        size = len(encoded.encode("utf-8"))
        if size > LARGE_THRESHOLD_BYTES:
            ptr_uuid = uuid.uuid4().hex
            redis_key = _redis_key(workspace_id, instance_id, ptr_uuid)
            await redis.set(redis_key, encoded, ex=TTL_SECONDS)
            out[key] = f"{POINTER_PREFIX}{ptr_uuid}"
        else:
            out[key] = value
    return out


async def read_state_with_pointers(
    state: Any,
    *,
    workspace_id: UUID,
    instance_id: UUID,
    redis: Redis,
) -> Any:
    """递归扫描 state，遇到 pointer 字符串从 Redis 拉回真实值。
    
    嵌套 dict / list 全部递归。pointer 不存在则替换为 {__ptr_missing__: <pointer>}（不抛错）。
    """
    if is_pointer(state):
        ptr_uuid = parse_pointer(state)
        redis_key = _redis_key(workspace_id, instance_id, ptr_uuid)
        encoded = await redis.get(redis_key)
        if encoded is None:
            return {"__ptr_missing__": state}
        return json.loads(encoded if isinstance(encoded, str) else encoded.decode("utf-8"))
    if isinstance(state, dict):
        return {k: await read_state_with_pointers(v, workspace_id=workspace_id, instance_id=instance_id, redis=redis)
                for k, v in state.items()}
    if isinstance(state, list):
        return [await read_state_with_pointers(v, workspace_id=workspace_id, instance_id=instance_id, redis=redis)
                for v in state]
    return state


async def cleanup_instance_pointers(
    workspace_id: UUID, instance_id: UUID, *, redis: Redis,
) -> int:
    """删除某实例的所有 pointers（Phase 7 自动清理使用）。"""
    pattern = f"{REDIS_KEY_PREFIX}:{workspace_id}:{instance_id}:*"
    deleted = 0
    async for key in redis.scan_iter(match=pattern, count=100):
        await redis.delete(key)
        deleted += 1
    return deleted
```

import os
from redis.asyncio import Redis, ConnectionPool

_pool: ConnectionPool | None = None

def get_redis_pool() -> ConnectionPool:
    global _pool
    if _pool is None:
        url = os.environ.get("REDIS_URL", "redis://localhost:6379/0")
        _pool = ConnectionPool.from_url(url, decode_responses=True, max_connections=20)
    return _pool

def get_redis() -> Redis:
    return Redis(connection_pool=get_redis_pool())
```
