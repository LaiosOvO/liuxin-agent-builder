export type NodeType = "start" | "end" | "llm" | "tool" | "if_else";

export interface DSLNode {
  id: string;
  type: NodeType;
  position: { x: number; y: number };
  config: Record<string, unknown>;
}

export interface DSLEdge {
  id: string;
  source: string;
  target: string;
}

export interface DSL {
  version: "1.0";
  name: string;
  state_schema: Record<string, "str" | "int" | "float" | "bool" | "list" | "dict" | "any">;
  nodes: DSLNode[];
  edges: DSLEdge[];
}

export const DSL_VERSION = "1.0";

export const RESERVED_NODE_IDS = new Set([
  "state", "event", "ptr", "__ptr__", "start_event", "end_event",
]);

export const NODE_ID_REGEX = /^[a-z][a-z0-9_]*$/;

```typescript
// web/src/lib/stores/canvas-store.ts
import { create } from "zustand";
import { addEdge, applyEdgeChanges, applyNodeChanges } from "@xyflow/react";
import type { Edge, EdgeChange, Node, NodeChange, Connection } from "@xyflow/react";
import type { DSL, DSLNode, DSLEdge } from "@/lib/types/dsl";

interface CanvasState {
  workflowId: string | null;
  workflowName: string;
  stateSchema: DSL["state_schema"];
  nodes: Node[];
  edges: Edge[];
  selectedNodeId: string | null;
  selectedEdgeId: string | null;

  // 加载/导出
  loadDSL: (dsl: DSL) => void;
  exportDSL: () => DSL;

  // 节点/边变更
  onNodesChange: (changes: NodeChange[]) => void;
  onEdgesChange: (changes: EdgeChange[]) => void;
  onConnect: (connection: Connection) => void;

  // 拖拽新节点入画布
  addNode: (type: DSLNode["type"], position: { x: number; y: number }) => void;
  
  // 配置面板更新
  updateNodeConfig: (nodeId: string, newConfig: Record<string, unknown>) => void;
  renameNode: (oldId: string, newId: string) => void;
  
  // 选中
  selectNode: (id: string | null) => void;
  selectEdge: (id: string | null) => void;
  
  // 工作流元信息
  setWorkflowName: (name: string) => void;
  updateStateSchema: (schema: DSL["state_schema"]) => void;
}

export const useCanvasStore = create<CanvasState>((set, get) => ({
  workflowId: null,
  workflowName: "新工作流",
  stateSchema: {},
  nodes: [],
  edges: [],
  selectedNodeId: null,
  selectedEdgeId: null,

  loadDSL: (dsl) => {
    const { nodes, edges } = dslToFlow(dsl);
    set({ workflowName: dsl.name, stateSchema: dsl.state_schema, nodes, edges });
  },
  exportDSL: () => flowToDsl(get().nodes, get().edges, get().workflowName, get().stateSchema),

  onNodesChange: (changes) => set({ nodes: applyNodeChanges(changes, get().nodes) }),
  onEdgesChange: (changes) => set({ edges: applyEdgeChanges(changes, get().edges) }),
  onConnect: (connection) => set({ edges: addEdge({ ...connection, id: nanoid() }, get().edges) }),

  addNode: (type, position) => {
    const newId = generateNodeId(type, get().nodes);
    set({
      nodes: [...get().nodes, {
        id: newId,
        type, // 自定义节点 type 名
        position,
        data: { config: defaultConfigFor(type), label: defaultLabelFor(type) },
      }],
    });
  },

  updateNodeConfig: (nodeId, newConfig) => set({
    nodes: get().nodes.map(n => n.id === nodeId ? { ...n, data: { ...n.data, config: newConfig } } : n),
  }),
  renameNode: (oldId, newId) => {
    // 修改节点 ID + 修改所有相关 edge 的 source/target
    ...
  },
  selectNode: (id) => set({ selectedNodeId: id, selectedEdgeId: null }),
  selectEdge: (id) => set({ selectedEdgeId: id, selectedNodeId: null }),
  setWorkflowName: (name) => set({ workflowName: name }),
  updateStateSchema: (schema) => set({ stateSchema: schema }),
}));
```

      const nodes: Node[] = dsl.nodes.map(n => ({
        id: n.id,
        type: n.type,  // 必须与 nodeTypes 注册名对齐
        position: n.position,
        data: { config: n.config, label: n.id },
      }));
      const edges: Edge[] = dsl.edges.map(e => ({

      nodes: Node[], edges: Edge[], name: string, state_schema: DSL["state_schema"]

      return {
        version: "1.0",
        name,
        state_schema,
        nodes: nodes.map(n => ({
          id: n.id,
          type: n.type as DSLNode["type"],
          position: n.position,
          config: (n.data?.config ?? {}) as Record<string, unknown>,
        })),
        edges: edges.map(e => ({ id: e.id, source: e.source, target: e.target })),
      };
    }
