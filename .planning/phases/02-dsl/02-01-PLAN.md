```sql
-- workflows: 工作流主表
CREATE TABLE workflows (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(120) NOT NULL,
    description TEXT,
    status VARCHAR(20) NOT NULL DEFAULT 'draft', -- draft / published / archived
    current_draft_version_id UUID,
    current_published_version_id UUID,
    created_by UUID NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX ix_workflows_ws_status_created ON workflows (workspace_id, status, created_at DESC);

-- workflow_versions: 版本快照（草稿/发布）
CREATE TABLE workflow_versions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    workflow_id UUID NOT NULL REFERENCES workflows(id) ON DELETE CASCADE,
    version_no INTEGER NOT NULL,            -- 草稿 = 0；发布 1 / 2 / 3...
    kind VARCHAR(10) NOT NULL,              -- 'draft' | 'published'
    dsl JSONB NOT NULL,                     -- 完整 DSL JSON
    created_by UUID NOT NULL REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (workflow_id, version_no, kind)
);
CREATE INDEX ix_workflow_versions_ws_wf ON workflow_versions (workspace_id, workflow_id);

-- flow_instances: 实例（运行/暂停/已完成/失败/中止）
CREATE TABLE flow_instances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    workflow_id UUID NOT NULL REFERENCES workflows(id),
    workflow_version_id UUID NOT NULL REFERENCES workflow_versions(id),
    dsl_snapshot JSONB NOT NULL,            -- 实例创建时锁定的 DSL 副本
    thread_id VARCHAR(120) NOT NULL UNIQUE, -- workspace_id:instance_id 格式
    status VARCHAR(20) NOT NULL DEFAULT 'pending', -- pending/running/completed/failed/aborted
    initial_input JSONB,
    final_output JSONB,
    error_message TEXT,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ
);
CREATE INDEX ix_flow_instances_ws_wf_status_created ON flow_instances (workspace_id, workflow_id, status, created_at DESC);
CREATE INDEX ix_flow_instances_ws_status_created ON flow_instances (workspace_id, status, created_at DESC);

-- node_states: 每个节点的执行状态
CREATE TABLE node_states (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    instance_id UUID NOT NULL REFERENCES flow_instances(id) ON DELETE CASCADE,
    node_id VARCHAR(80) NOT NULL,           -- DSL node id（如 'llm_1'）
    node_type VARCHAR(20) NOT NULL,         -- start/end/llm/tool/if_else
    status VARCHAR(20) NOT NULL,            -- pending/running/completed/failed
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    error_message TEXT,
    retries INTEGER NOT NULL DEFAULT 0,
    output_summary JSONB                    -- 不存 raw output（走 Redis pointer）
);
CREATE INDEX ix_node_states_ws_instance_created ON node_states (workspace_id, instance_id, started_at);
```

    ```python
    from typing import TypedDict, get_type_hints
    
    PY_TYPE_MAP = {
        "str": str, "int": int, "float": float, "bool": bool,
        "list": list, "dict": dict, "any": object,
    }
    
    def build_state_typeddict(state_schema: dict[str, str], name: str = "WorkflowState") -> type:
        """从 DSL state_schema dict 动态构造 TypedDict 类。
        
        Args:
            state_schema: {"field_name": "str" | "int" | ...}
            name: 生成的 TypedDict 类名
        
        Returns:
            一个 TypedDict 子类（用作 StateGraph(StateType)）
        
        Raises:
            ValueError: 包含未知 Python 类型字符串
        """
        unknown = [t for t in state_schema.values() if t not in PY_TYPE_MAP]
        if unknown:
            raise ValueError(f"未知类型: {unknown}（支持: {list(PY_TYPE_MAP)})")
        fields = {field: PY_TYPE_MAP[type_str] for field, type_str in state_schema.items()}
        return TypedDict(name, fields, total=False)  # type: ignore[misc]
    
    SCHEMA_TYPES = list(PY_TYPE_MAP.keys())
    ```
