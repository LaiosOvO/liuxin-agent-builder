```python
# hitl_action_service.py 改造点（伪代码）

async def submit_action(self, *, payload, action, reason, form_data, ip, ua):
    # ── 1. 加载 flow_instance（不变）─────────────────────
    # ── 2. advisory_xact_lock（不变）─────────────────────
    # ── 3. 加载 node_state + form_schema 校验（不变）──────
    # ── 4. consume jti（不变）────────────────────────────
    # ── 5. invalidate_siblings（不变 — 同节点其他 token）─

    # ── 6. NEW: chain 推进分叉 ───────────────────────────
    actor_id = UUID(payload["actor_id"])
    chain_result = compute_chain_advance(ns.payload or {}, actor_id, action)

    # 6a. parallel_* 模式终止 → invalidate_chain
    invalidated_chain: list[tuple[UUID, UUID]] = []
    if chain_result.invalidate_others:
        invalidated_chain = await store.invalidate_chain(
            instance_id=instance_id,
            except_jti=jti,
        )
        # 入队补通知（"已被 X 处理 / 已被 X 拒绝"）
        await self._supplement_notify(
            invalidated_chain, action=action, actor_email=actor_email,
        )

    # 6b. sequential approve 推进 → 创建下一人 token + 发通知
    if chain_result.next_approvers:
        hitl_svc = HitlService(self.db)
        new_tokens = await hitl_svc.batch_create_tokens_for_actors(
            instance_id=instance_id,
            node_state_id=node_state_id,
            actor_ids=chain_result.next_approvers,
            allowed_actions=["approve", "return", "reject"],  # chain 中所有非首发审批人无 submit 权
        )
        # 入队通知（沿用 Phase 3 enqueue_hitl_email）
        await self._enqueue_chain_notifications(new_tokens, ns, chain_result)

    # ── 7. 写 node_state.payload + status（用 chain_result）──
    ns.payload = chain_result.new_payload
    ns.status = chain_result.new_status

    # ── 8. LangGraph resume（不变；状态非 in_review 时 resume 推进 graph）─
    # ── 9. audit_log + commit（不变；audit_log meta 加 chain_mode + invalidated_count）

    # 结构化日志（Phase 7 Run Viewer 用）
    log.info(
        "hitl.chain.advance",
        extra={
            "chain_mode": ns.payload.get("approval_chain", {}).get("mode"),
            "actor_id": str(actor_id),
            "action": action,
            "new_status": chain_result.new_status,
            "next_approvers_count": len(chain_result.next_approvers),
            "invalidated_count": len(invalidated_chain),
            "instance_id": str(instance_id),
            "node_state_id": str(node_state_id),
        },
    )
```

```python
# hitl_service.py 新增方法

async def batch_create_tokens_for_actors(
    self,
    *,
    instance_id: UUID,
    node_state_id: UUID,
    actor_ids: list[UUID],
    allowed_actions: list[str],
    expires_in_seconds: int = DEFAULT_TOKEN_EXPIRES_IN,
) -> list[HitlToken]:
    """批量为多个 actor 创建 tokens（parallel_* 模式初始化 + sequential approve 推进用）

    Returns: list[HitlToken]（已 add_all + flush）

    每个 (actor_id × action) 一行 token，所以 len(returned) == len(actor_ids) × len(allowed_actions)
    """
    expires_at = datetime.now(timezone.utc) + timedelta(seconds=expires_in_seconds)
    tokens = [
        HitlToken(
            jti=uuid4(),
            instance_id=instance_id,
            node_state_id=node_state_id,
            actor_id=actor_id,
            action=action,
            expires_at=expires_at,
        )
        for actor_id in actor_ids
        for action in allowed_actions
    ]
    self.db.add_all(tokens)
    await self.db.flush()
    return tokens
```
