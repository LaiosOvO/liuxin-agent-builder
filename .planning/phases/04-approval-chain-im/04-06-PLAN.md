# backend/app/agent_builder/notification/cards/feishu_card.py

def build_feishu_hitl_card(
    *,
    flow_title: str,
    node_title: str,
    applicant_name: str,
    actor_name: str,
    deadline_at: str,
    description: str,
    deeplinks: list[dict[str, str]],  # [{"action":"approve","url":"..."}]
) -> dict:
    """构造飞书 Interactive Card 2.0 卡片 JSON

    参考 reading-im-sdk-04-06-feishu §2 卡片 schema
    """
    # 按钮颜色：approve=绿色（primary）/ return=灰色 / reject=红色（danger）
    action_color_map = {"approve": "primary", "return": "default", "reject": "danger"}
    action_label_map = {"approve": "同意", "return": "退回", "reject": "拒绝", "submit": "提交"}

    buttons = []
    for dl in deeplinks:
        action = dl["action"]
        buttons.append({
            "tag": "button",
            "text": {"tag": "plain_text", "content": action_label_map.get(action, action)},
            "type": action_color_map.get(action, "default"),
            "multi_url": {
                "url": dl["url"],
                "pc_url": dl["url"],
                "android_url": dl["url"],
                "ios_url": dl["url"],
            },
        })

    return {
        "config": {"wide_screen_mode": True},
        "header": {
            "title": {"tag": "plain_text", "content": f"📋 审批待办：{flow_title}"},
            "template": "blue",
        },
        "elements": [
            {
                "tag": "div",
                "fields": [
                    {"is_short": True, "text": {"tag": "lark_md", "content": f"**节点**\n{node_title}"}},
                    {"is_short": True, "text": {"tag": "lark_md", "content": f"**申请人**\n{applicant_name}"}},
                    {"is_short": True, "text": {"tag": "lark_md", "content": f"**审批人**\n{actor_name}"}},
                    {"is_short": True, "text": {"tag": "lark_md", "content": f"**截止时间**\n{deadline_at}"}},
                ],
            },
            {"tag": "div", "text": {"tag": "lark_md", "content": f"**详情**\n{description}"}},
            {"tag": "hr"},
            {"tag": "action", "actions": buttons},
        ],
    }


def build_feishu_processed_card(*, original_card: dict, processed_by: str) -> dict:
    """卡片决策后更新为只读 + 显示'已被 X 处理'"""
    # 复制 header + 加角标 + 移除 action block
    new_card = {**original_card}
    new_card["elements"] = [
        e for e in new_card["elements"] if e.get("tag") != "action"
    ]
    new_card["elements"].append({
        "tag": "note",
        "elements": [{"tag": "plain_text", "content": f"✓ 已被 {processed_by} 处理"}],
    })
    new_card["header"]["template"] = "grey"
    return new_card

# backend/app/agent_builder/notification/providers/feishu.py

from __future__ import annotations
import json
import logging
from typing import Any

import lark_oapi as lark
from lark_oapi.api.im.v1 import (
    CreateMessageRequest, CreateMessageRequestBody,
    PatchMessageRequest, PatchMessageRequestBody,
)

from app.agent_builder.notification.providers.base import IMProvider, PROVIDER_FEISHU
from app.agent_builder.notification.cards.feishu_card import (
    build_feishu_hitl_card, build_feishu_processed_card,
)

log = logging.getLogger(__name__)

# CLAUDE.md §3 强制版本校验（防 1.6.0-1.6.3 yanked 版本）
_EXPECTED_LARK_VERSION = "1.6.5"


class FeishuProvider:
    """飞书 Provider — 实现 IMProvider Protocol"""

    name = PROVIDER_FEISHU

    def __init__(self, app_id: str, app_secret: str) -> None:
        self.app_id = app_id
        self.app_secret = app_secret
        self._client: lark.Client | None = None
        # 启动校验 SDK 版本
        actual_version = getattr(lark, "__version__", "unknown")
        if actual_version != _EXPECTED_LARK_VERSION:
            log.warning(
                "lark-oapi 版本 %s != %s（CLAUDE.md §3 锁定）— 行为可能异常",
                actual_version, _EXPECTED_LARK_VERSION,
            )

    @property
    def client(self) -> lark.Client:
        """延迟初始化 client（避免 module import 时调用网络）"""
        if self._client is None:
            self._client = (
                lark.Client.builder()
                .app_id(self.app_id)
                .app_secret(self.app_secret)
                .log_level(lark.LogLevel.WARNING)
                .build()
            )
        return self._client

    async def send_hitl_card(
        self, *, recipient: str, flow_title: str, node_title: str,
        applicant_name: str, actor_name: str, deadline_at: str,
        description: str, deeplinks: list[dict[str, str]],
    ) -> dict[str, Any]:
        card = build_feishu_hitl_card(
            flow_title=flow_title, node_title=node_title,
            applicant_name=applicant_name, actor_name=actor_name,
            deadline_at=deadline_at, description=description,
            deeplinks=deeplinks,
        )
        request = (
            CreateMessageRequest.builder()
            .receive_id_type("open_id")  # recipient 是飞书 open_id
            .request_body(
                CreateMessageRequestBody.builder()
                .receive_id(recipient)
                .msg_type("interactive")
                .content(json.dumps(card))
                .build()
            )
            .build()
        )
        # lark-oapi 同步 client；要在 thread executor 跑
        import asyncio
        loop = asyncio.get_event_loop()
        response = await loop.run_in_executor(None, self.client.im.v1.message.create, request)

        if not response.success():
            err_msg = f"feishu send_hitl_card failed: code={response.code} msg={response.msg}"
            log.error(err_msg)
            # 把 SDK 错误映射为 ConnectionError 让 tenacity 重试
            raise ConnectionError(err_msg)

        return {
            "message_id": response.data.message_id,
            "raw_response": {"code": response.code, "msg": response.msg},
        }

    async def update_card(
        self, *, message_id: str, new_content: dict[str, Any],
    ) -> None:
        request = (
            PatchMessageRequest.builder()
            .message_id(message_id)
            .request_body(
                PatchMessageRequestBody.builder()
                .content(json.dumps(new_content))
                .build()
            )
            .build()
        )
        import asyncio
        loop = asyncio.get_event_loop()
        response = await loop.run_in_executor(None, self.client.im.v1.message.patch, request)
        if not response.success():
            err_msg = f"feishu update_card failed: code={response.code} msg={response.msg}"
            log.error(err_msg)
            raise ConnectionError(err_msg)

    async def send_supplement_text(
        self, *, recipient: str, text: str,
    ) -> None:
        request = (
            CreateMessageRequest.builder()
            .receive_id_type("open_id")
            .request_body(
                CreateMessageRequestBody.builder()
                .receive_id(recipient)
                .msg_type("text")
                .content(json.dumps({"text": text}))
                .build()
            )
            .build()
        )
        import asyncio
        loop = asyncio.get_event_loop()
        response = await loop.run_in_executor(None, self.client.im.v1.message.create, request)
        if not response.success():
            log.warning(
                "feishu send_supplement_text failed: code=%s msg=%s",
                response.code, response.msg,
            )
