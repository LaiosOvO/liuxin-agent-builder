```python
async def _on_hitl_enter(
    self, *, node_def: dict, instance_id: UUID, workspace_id: UUID,
    db: AsyncSession, redis,
) -> UUID:
    """HITL 节点首次 enter 时调（在 interrupt 之前）

    职责：
    1. 解析 node_def.config.chain_mode / assignees / notify_channels
    2. 创建 node_states 行 + 写 _node_state_id 注入 state
    3. build_initial_payload(chain_mode, approvers) + 写 node_state.payload
    4. 对 chain_mode 分发创建 token：
       - single / sequential：仅给 approvers[0] 创建 token
       - parallel_all / parallel_any：给所有 approvers 创建 token（batch_create_tokens_for_actors）
    5. 调 NotificationService.enqueue_hitl_multichannel(channels=notify_channels) 发通知
    6. 结构化日志 'hitl.node.entered'

    Returns:
        node_state_id（写入 state['_node_state_id']）
    """
    config = node_def.get("config", {})
    chain_mode = config.get("chain_mode", "single")
    assignees = config.get("assignees") or []
    notify_channels = config.get("notify_channels", ["email"])
    timeout_seconds = config.get("timeout_seconds", 86400)
    form_schema = config.get("form_schema") or {}

    # 1. 解析 assignees → UUID list（email / user: / role: 表达式，复用 escalation resolver 思路）
    approver_uuids: list[UUID] = await self._resolve_assignees(assignees, workspace_id)
    if not approver_uuids:
        raise NodeExecutionError(node_def["id"], "无法解析 assignees → 任何 user_id")

    # 2. 创建 node_state 行
    node_state = NodeState(
        workspace_id=workspace_id, instance_id=instance_id,
        node_id=node_def["id"], node_type="hitl",
        status="waiting_human", started_at=datetime.now(UTC),
    )

    # 3. 初始 payload（含 chain 字段）
    first_actor = await db.get(User, approver_uuids[0])
    node_state.payload = build_initial_payload(
        current_actor_id=approver_uuids[0],
        current_actor_email=first_actor.email if first_actor else "",
        role="reviewer",  # chain 内全是 reviewer phase（不再有 submit phase 区分）
        timeout_seconds=timeout_seconds,
        form_schema=form_schema,
        approvers=approver_uuids,
        chain_mode=chain_mode,  # 04-01 已扩展 build_initial_payload
    )
    db.add(node_state)
    await db.flush()

    # 4. 创建 token（按 chain_mode 分发）
    hitl_svc = HitlService(db)
    if chain_mode in ("single", "sequential"):
        target_actors = [approver_uuids[0]]
    else:  # parallel_all / parallel_any
        target_actors = approver_uuids

    tokens = await hitl_svc.batch_create_tokens_for_actors(
        instance_id=instance_id, node_state_id=node_state.id,
        actor_ids=target_actors,
        allowed_actions=["approve", "return", "reject"],
        expires_in_seconds=timeout_seconds,
    )
    await db.commit()

    # 5. 发通知（按 channels fan-out + 按 actor fan-out）
    notif_svc = NotificationService(db, arq_pool=self.arq_pool)
    flow_instance = await db.get(FlowInstance, instance_id)
    for actor_id in target_actors:
        actor = await db.get(User, actor_id)
        if actor is None:
            continue
        actor_tokens = [t for t in tokens if t.actor_id == actor_id]
        await notif_svc.enqueue_hitl_multichannel(
            workspace_id=workspace_id, instance_id=instance_id,
            node_state_id=node_state.id,
            recipient_email=actor.email,
            recipient_im_bindings=actor.im_bindings or {},
            tokens=actor_tokens,
            form_schema=form_schema,
            deadline_at=datetime.now(UTC) + timedelta(seconds=timeout_seconds),
            actor_name=actor.full_name or actor.email,
            flow_title=flow_instance.title if flow_instance else "",
            node_title=node_def.get("name", node_def["id"]),
            applicant_name="...",  # 从 state 提取或 DSL config
            description=config.get("description", ""),
            channels=notify_channels,
        )

    # 6. 结构化日志
    log.info(
        "hitl.node.entered",
        extra={
            "chain_mode": chain_mode,
            "approvers_count": len(approver_uuids),
            "notify_channels": notify_channels,
            "instance_id": str(instance_id),
            "node_state_id": str(node_state.id),
        },
    )

    return node_state.id
```
