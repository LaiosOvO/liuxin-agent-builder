```python
# 主路径：template_card button_interaction
def build_wecom_template_card(
    *, flow_title, node_title, applicant_name, actor_name,
    deadline_at, description, deeplinks,
) -> dict:
    """构造企微 button_interaction template_card

    参考企微文档 template_card.button_interaction
    """
    buttons = [
        {
            "text": _label_for(dl["action"]),
            "style": _style_for(dl["action"]),  # 1=蓝色（primary）2=绿色 3=灰色 4=红色
            "key": _build_button_key(dl),
            # button 不直接含 URL — 用 task_id + key 触发回调；
            # Phase 4 简化：用 jump_list 字段（点击跳 URL）
            "jump_list": [{"type": 1, "url": dl["url"], "title": _label_for(dl["action"])}],
        }
        for dl in deeplinks
    ]
    return {
        "card_type": "button_interaction",
        "source": {"icon_url": "...", "desc": "agent-builder"},
        "main_title": {"title": f"审批待办：{flow_title}", "desc": node_title},
        "horizontal_content_list": [
            {"keyname": "申请人", "value": applicant_name},
            {"keyname": "审批人", "value": actor_name},
            {"keyname": "截止时间", "value": deadline_at},
        ],
        "card_action": {"type": 1, "url": deeplinks[0]["url"] if deeplinks else ""},
        "button_selection": {"question_key": "approval_action", "title": "请决策", "option_list": []},
        "button_list": buttons,
    }


# Fallback: Bot Webhook Markdown
def build_wecom_webhook_markdown(
    *, flow_title, node_title, applicant_name, deadline_at,
    description, deeplinks,
) -> dict:
    """构造企微 Bot Webhook markdown 消息（fallback）"""
    lines = [
        f"### 审批待办：{flow_title}",
        f"**节点**: {node_title}",
        f"**申请人**: {applicant_name}",
        f"**截止时间**: {deadline_at}",
        f"**详情**: {description}",
        "",
        "**请决策**:",
    ]
    for dl in deeplinks:
        lines.append(f"- [{_label_for(dl['action'])}]({dl['url']})")
    return {"msgtype": "markdown", "markdown": {"content": "\n".join(lines)}}


def _label_for(action: str) -> str:
    return {"approve": "同意", "return": "退回", "reject": "拒绝", "submit": "提交"}.get(action, action)


def _style_for(action: str) -> int:
    return {"approve": 2, "return": 3, "reject": 4}.get(action, 1)
```

```python
class WeComProvider:
    name = PROVIDER_WECOM

    def __init__(
        self, *, corp_id: str, agent_id: str, secret: str,
        bot_webhook_key: str | None = None,

        self.corp_id = corp_id
        self.agent_id = agent_id
        self.secret = secret

        self.use_bot_fallback = use_bot_fallback
        self._client = None

    async def send_hitl_card(self, *, recipient, ..., deeplinks):
        if self.use_bot_fallback:
            return await self._send_via_bot_webhook(deeplinks=deeplinks, ...)

        # httpx POST 路径
        url = f"https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key={self.bot_webhook_key}"
```
