```python
# backend/app/agent_builder/workflow/hitl_payload.py 新增部分

from dataclasses import dataclass, field
from typing import Literal

ChainMode = Literal["single", "sequential", "parallel_all", "parallel_any"]

@dataclass(frozen=True)
class ChainAdvanceResult:
    """compute_chain_advance 返回值（不可变）

    Attributes:
        new_status: 节点新 status（in_review/done/rejected/returned）
        new_payload: 新 payload dict（immutable，调用方写入 node_state.payload）
        next_approvers: 下轮需创建 token 的人（sequential approve 给 1 人；其他情况空 list）
        invalidate_others: 是否需调 invalidate_chain（parallel_* reject 或 parallel_any approve）
        supplement_notify: 需补"已终止/已处理"通知的 actor_id 列表
    """
    new_status: NodeStatus
    new_payload: dict[str, Any]
    next_approvers: list[UUID] = field(default_factory=list)
    invalidate_others: bool = False
    supplement_notify: list[UUID] = field(default_factory=list)


def compute_chain_advance(
    payload: dict[str, Any],
    actor_id: UUID,
    action: Action,
) -> ChainAdvanceResult:
    """计算审批链推进结果（纯函数，不修改入参）

    决策依据 04-RESEARCH.md §二 状态机表格 + 04-CONTEXT.md §审批链 4 模式语义。

    Raises:
        ValueError: 非法 actor（不在 approval_chain.approvers 内）或非法 action
    """
    chain = payload.get("approval_chain") or {}
    mode: ChainMode = chain.get("mode", "single")
    approvers = [UUID(a) for a in chain.get("approvers", [])]
    if actor_id not in approvers:
        raise ValueError(f"actor_id={actor_id} 不在 approvers={approvers} 内")

    if mode == "single":
        return _advance_single(payload, action)
    if mode == "sequential":
        return _advance_sequential(payload, actor_id, action)
    if mode == "parallel_all":
        return _advance_parallel_all(payload, actor_id, action)
    if mode == "parallel_any":
        return _advance_parallel_any(payload, actor_id, action)
    raise ValueError(f"非法 chain_mode={mode}")

# 4 个内部 helper：_advance_sequential / _advance_parallel_all / _advance_parallel_any / _advance_single
# 每个返回 ChainAdvanceResult；保证 immutability（用 {**payload, ...} dict copy + new list）
```

```python
# backend/app/agent_builder/workflow/hitl_token_store.py 新增方法

async def invalidate_chain(
    self,
    instance_id: UUID,
    except_jti: UUID,
) -> list[tuple[UUID, UUID]]:
    """跨整个 instance 失效所有未消费 token（parallel_* reject / parallel_any approve 用）

    必须在 advisory_lock 内调（Pitfall 2 防护）— 由 service 层保证。

    Returns:
        list of (jti, actor_id) — 用于发"已被处理"补通知

    与 invalidate_siblings 的区别：
    - invalidate_siblings(node_state_id, except_jti) — 单节点内
    - invalidate_chain(instance_id, except_jti) — 整个 instance（跨节点）
    """
    from datetime import datetime, timezone
    stmt = (
        update(HitlToken)
        .where(
            HitlToken.instance_id == instance_id,
            HitlToken.jti != except_jti,
            HitlToken.used_at.is_(None),
        )
        .values(
            used_at=datetime.now(timezone.utc),
            used_ip="system:chain-invalidate",
            used_ua="system:invalidate_chain",
        )
        .returning(HitlToken.jti, HitlToken.actor_id)
    )
    result = await self.db.execute(stmt)
    rows = list(result.all())  # [(jti, actor_id), ...]
    if rows:
        pipe = self.redis.pipeline()
        for jti, _ in rows:
            pipe.set(f"{_REDIS_PREFIX}{jti}", "consumed", ex=_REDIS_TTL)
        await pipe.execute()
    return rows
```
