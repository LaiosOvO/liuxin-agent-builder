```python
# notification_service.py 新增

async def enqueue_hitl_multichannel(
    self,
    *,
    workspace_id: UUID,
    instance_id: UUID,
    node_state_id: UUID,
    recipient_email: str,
    recipient_im_bindings: dict[str, str] | None,  # {"feishu": "ou_abc", "wecom": "WuPing"}
    tokens: Sequence[HitlToken],
    form_schema: dict,
    deadline_at: datetime,
    actor_name: str,
    flow_title: str,
    node_title: str,
    applicant_name: str,
    description: str,
    channels: list[str],         # ["email", "feishu", "wecom"]
    reminder_round: int = 0,
) -> list[Notification]:
    """fan-out 入队：每个 channel 写 1 行 + enqueue 1 个 job

    Returns:
        list[Notification]（已 commit），每行对应一个 channel

    每个 channel 对 recipient 解析：
    - email：用 recipient_email
    - feishu / wecom / dingtalk / slack / mattermost：用 recipient_im_bindings[channel]
      若未配置 → 跳过该 channel（warn 日志）
    """
    created: list[Notification] = []
    for channel in channels:
        if channel == "email":
            recipient = recipient_email
        else:
            recipient = (recipient_im_bindings or {}).get(channel)
            if not recipient:
                log.warning(
                    "im_bindings 缺 %s（actor=%s instance=%s）— 跳过该 channel",
                    channel, recipient_email, instance_id,
                )
                continue

        # 构造 payload（含 tokens / form_schema / deadline_at / titles / description）
        payload = self._build_hitl_payload(
            tokens=tokens, form_schema=form_schema, deadline_at=deadline_at,
            actor_name=actor_name, flow_title=flow_title, node_title=node_title,
            applicant_name=applicant_name, description=description,
        )

        notif = Notification(
            workspace_id=workspace_id, instance_id=instance_id,
            node_state_id=node_state_id, channel=channel, recipient=recipient,
            reminder_round=reminder_round, status="pending", payload=payload,
        )
        self.db.add(notif)
        created.append(notif)

    await self.db.flush()
    await self.db.commit()
    for n in created:
        await self.db.refresh(n)

    # commit 之后才 enqueue（防 worker 抢跑事务未提交的行）
    for notif in created:
        job_name = "send_hitl_email_job" if notif.channel == "email" else "send_hitl_card_job"
        if self.arq is not None:
            await self.arq.enqueue_job(job_name, str(notif.id))
        else:
            await self._fallback_dispatch(notif, job_name)

    return created


async def enqueue_generic_im_card(
    self,
    *,
    workspace_id, instance_id, node_state_id, recipient,  # recipient = im_user_id
    channel: str, subject: str, body: str,
) -> Notification:
    """Notification 节点 IM 通道入口（与 enqueue_generic_email 平行）"""
    payload = {
        "generic": True,
        "subject": subject, "body": body, "recipient_im": recipient,
    }
    notif = Notification(
        workspace_id=workspace_id, instance_id=instance_id,
        node_state_id=node_state_id, channel=channel, recipient=recipient,
        reminder_round=0, status="pending", payload=payload,
    )
    self.db.add(notif)
    await self.db.flush()
    await self.db.commit()
    await self.db.refresh(notif)

    if self.arq is not None:
        await self.arq.enqueue_job("send_hitl_card_job", str(notif.id))
    return notif
```

```python
        node_state_id = await self._resolve_node_state_id(db)

        for channel in channels:
            if channel == "email":
                # 现有逻辑（email 校验 + enqueue_generic_email）
                ...
            else:
                # IM channel — recipient 是 im_user_id 字符串
                for recipient in self._normalize_recipients(recipients, channel):
                    try:
                        notif = await svc.enqueue_generic_im_card(
                            workspace_id=self.workspace_id,
                            instance_id=self.instance_id,
                            node_state_id=node_state_id,
                            recipient=recipient,
                            channel=channel,
                            subject=config.get("subject", "通知"),
                            body=config.get("body", ""),
                        )
                        notification_ids.append(notif.id)
                        sent_count += 1
                    except Exception as e:
                        log.exception("IM 入队失败 channel=%s", channel)
                        await db.rollback()
                        failed_count += 1

    return {"sent_count": sent_count, "failed_count": failed_count, "notification_ids": notification_ids}
```
