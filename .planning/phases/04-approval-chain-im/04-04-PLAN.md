# escalation_service.py 修改 resolve_escalate_to 签名 + 返回类型

class EscalationExprError(Exception):
    """超时升级表达式解析错误（无法解析 → 配置问题）"""

async def resolve_escalate_to(
    self,
    *,
    node_config: dict[str, Any] | None,
    workspace_id: UUID,
) -> list[str] | None:
    """解析升级人 email 列表（Phase 4 4 表达式）

    Returns:
        list[email]：可能 1 个或多个 email（role: 可匹配多人）
        None：表达式空 + fallback 也无 admin

    Raises:
        NotImplementedError: dept: 表达式（Phase 5 实现）
    """
    if not node_config:
        # 直接走 fallback（保持现有行为）
        emails = await self._fallback_workspace_admin_emails(workspace_id)
        return emails or None

    expr = node_config.get("escalate_to")
    if not expr or not isinstance(expr, str):
        emails = await self._fallback_workspace_admin_emails(workspace_id)
        return emails or None

    expr = expr.strip()

    # 1. dept:<name> — Phase 5 实现
    if expr.startswith("dept:"):
        raise NotImplementedError(
            f"dept: 表达式（{expr}）将于 Phase 5（IM 目录双向同步）实现"
        )

    # 2. user:<uuid>
    if expr.startswith("user:"):
        try:
            uid = UUID(expr[5:].strip())
        except ValueError:
            log.warning("resolve_escalate_to: 非法 user: 表达式 %s", expr)
            return None
        email = await self._get_user_email(uid, workspace_id)
        return [email] if email else None

    # 3. role:<code>
    if expr.startswith("role:"):
        role_code = expr[5:].strip()
        emails = await self._get_emails_by_role(role_code, workspace_id)
        return emails or None

    # 4. email（默认 — 含 @）
    if "@" in expr and ":" not in expr:
        return [expr]

    # 都不匹配 → fallback
    log.warning("resolve_escalate_to: 未识别表达式 %s，走 fallback", expr)
    emails = await self._fallback_workspace_admin_emails(workspace_id)
    return emails or None


async def _get_user_email(self, user_id: UUID, workspace_id: UUID) -> str | None:
    """查 user_id 用户的 email，必须属于 workspace_id"""
    stmt = (
        select(User.email)
        .join(UserWorkspaceRole, UserWorkspaceRole.user_id == User.id)
        .where(
            User.id == user_id,
            UserWorkspaceRole.workspace_id == workspace_id,
            User.status == "active",
        )
        .limit(1)
    )
    result = await self.db.execute(stmt)
    return result.scalar_one_or_none()


async def _get_emails_by_role(self, role_code: str, workspace_id: UUID) -> list[str]:
    """查 workspace 内具有 role_code 的所有 active 用户 email"""
    stmt = (
        select(User.email)
        .join(UserWorkspaceRole, UserWorkspaceRole.user_id == User.id)
        .join(Role, Role.id == UserWorkspaceRole.role_id)
        .where(
            UserWorkspaceRole.workspace_id == workspace_id,
            Role.code == role_code,
            User.status == "active",
        )
        .distinct()
    )
    result = await self.db.execute(stmt)
    return [str(e) for e in result.scalars().all()]

# perform_escalation 改造：接受 list[str] 升级人

async def perform_escalation(self, ns: NodeState) -> None:
    flow_instance = await self.db.get(FlowInstance, ns.instance_id)
    if flow_instance is None:
        return

    node_config = self._extract_node_config(flow_instance, ns.node_id)

    try:
        escalate_emails = await self.resolve_escalate_to(
            node_config=node_config,
            workspace_id=ns.workspace_id,
        )
    except NotImplementedError as e:
        log.error("perform_escalation: %s（节点 %s）", e, ns.id)
        # Phase 4 仅 email/user/role 支持 — dept: 配置错误就跳过升级
        return

    if not escalate_emails:
        log.error("perform_escalation: 无升级人，跳过")
        return

    # 写 records 加 escalate 记录（不变）+ 多人写 escalate_emails 列表
    now = datetime.now(timezone.utc)
    old_payload = ns.payload or {}
    new_record = {
        # ... 现有字段
        "escalate_to": escalate_emails,         # 改为 list
        "escalate_count": len(escalate_emails),
    }
    new_payload = {
        **old_payload,
        "records": [*(old_payload.get("records") or []), new_record],
        "escalate_to": escalate_emails,           # 改为 list
    }
    ns.payload = new_payload

    # 对每个 email 独立发邮件 + audit_log
    for email in escalate_emails:
        await self._send_escalation_email(ns, email, old_payload)
        await self._write_audit_log(ns, email)

    # 结构化日志
    log.info(
        "hitl.escalation.resolved",
        extra={
            "expression": (node_config or {}).get("escalate_to"),
            "matched_count": len(escalate_emails),
            "workspace_id": str(ns.workspace_id),
            "node_state_id": str(ns.id),
        },
    )
