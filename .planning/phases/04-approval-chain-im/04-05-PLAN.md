# backend/app/agent_builder/notification/providers/base.py

from typing import Protocol, runtime_checkable, Any
from abc import abstractmethod

# Provider name 常量（避免字符串散布）
PROVIDER_FEISHU = "feishu"
PROVIDER_WECOM = "wecom"
PROVIDER_DINGTALK = "dingtalk"
PROVIDER_SLACK = "slack"
PROVIDER_MATTERMOST = "mattermost"

KNOWN_PROVIDERS: frozenset[str] = frozenset({
    PROVIDER_FEISHU, PROVIDER_WECOM, PROVIDER_DINGTALK,
    PROVIDER_SLACK, PROVIDER_MATTERMOST,
})


@runtime_checkable
class IMProvider(Protocol):
    """IM 出站投递接口。

    Phase 4：仅实现 send_hitl_card / update_card / send_supplement_text
    Phase 4.5 扩展：subscribe / verify_webhook_signature（入站）— 当前 Phase 不强制实现
    """

    name: str  # PROVIDER_FEISHU / PROVIDER_WECOM / ...

    async def send_hitl_card(
        self,
        *,
        recipient: str,                  # IM user_id（非 email）
        flow_title: str,
        node_title: str,
        applicant_name: str,
        actor_name: str,
        deadline_at: str,
        description: str,
        deeplinks: list[dict[str, str]], # [{"action":"approve","url":"..."}]
    ) -> dict[str, Any]:
        """投递卡片，返回 {message_id, raw_response}"""
        ...

    async def update_card(
        self,
        *,
        message_id: str,
        new_content: dict[str, Any],
    ) -> None:
        """更新卡片（部分 provider 不支持，应抛 NotImplementedError）"""
        ...

    async def send_supplement_text(
        self,
        *,
        recipient: str,
        text: str,
    ) -> None:
        """补发文本（"已被 X 处理"，对不支持 update_card 的 provider 用）"""
        ...

    # ── Phase 4.5 预留（Phase 4 不实现 → 默认抛 NotImplementedError）─────────

    async def subscribe(self, on_event: Any) -> None:
        """Phase 4.5 入站事件订阅 — 当前 Phase raise NotImplementedError"""
        raise NotImplementedError(f"{self.name} subscribe 将于 Phase 4.5 实现")

    async def verify_webhook_signature(
        self, headers: dict[str, str], body: bytes,
    ) -> bool:
        """Phase 4.5 webhook 签名校验 — 当前 Phase raise NotImplementedError"""
        raise NotImplementedError(f"{self.name} verify_webhook_signature 将于 Phase 4.5 实现")


# ── ProviderRegistry ──────────────────────────────────────────────────────

_PROVIDERS: dict[str, IMProvider] = {}


def register_provider(provider: IMProvider) -> None:
    """注册 provider（FastAPI lifespan startup 时调）"""
    _PROVIDERS[provider.name] = provider


def get_provider(name: str) -> IMProvider:
    """获取 provider；未注册抛 KeyError"""
    if name not in _PROVIDERS:
        raise KeyError(f"IM provider '{name}' 未注册（注册的: {list(_PROVIDERS.keys())}）")
    return _PROVIDERS[name]


def list_providers() -> list[str]:
    return list(_PROVIDERS.keys())


def clear_providers() -> None:
    """测试用 fixture：每个测试 setup/teardown 调"""
    _PROVIDERS.clear()

# backend/app/agent_builder/notification/providers/mock.py

from dataclasses import dataclass, field
from typing import Any

@dataclass
class MockCallRecord:
    """Mock provider 调用记录（E2E + 集成测试断言用）"""
    method: str  # 'send_hitl_card' / 'update_card' / 'send_supplement_text'
    recipient: str
    payload: dict[str, Any]


class MockIMProvider:
    """测试用 Provider — 不发真实 IM 请求，仅记录调用"""

    def __init__(self, name: str) -> None:
        self.name = name
        self.calls: list[MockCallRecord] = []
        self.fail_send_hitl_card: bool = False  # 测试用：true 时 send_hitl_card 抛错

    async def send_hitl_card(self, *, recipient, flow_title, node_title,
                             applicant_name, actor_name, deadline_at,
                             description, deeplinks) -> dict[str, Any]:
        if self.fail_send_hitl_card:
            raise ConnectionError(f"mock {self.name} simulated network error")
        rec = MockCallRecord(
            method="send_hitl_card",
            recipient=recipient,
            payload={
                "flow_title": flow_title, "node_title": node_title,
                "applicant_name": applicant_name, "actor_name": actor_name,
                "deadline_at": deadline_at, "description": description,
                "deeplinks": list(deeplinks),
            },
        )
        self.calls.append(rec)
        return {"message_id": f"mock-msg-{len(self.calls)}", "raw_response": {}}

    async def update_card(self, *, message_id, new_content) -> None:
        self.calls.append(MockCallRecord(
            method="update_card", recipient="", payload={
                "message_id": message_id, "new_content": new_content,
            },
        ))

    async def send_supplement_text(self, *, recipient, text) -> None:
        self.calls.append(MockCallRecord(
            method="send_supplement_text", recipient=recipient,
            payload={"text": text},
        ))
