# backend/app/agent_builder/notification/cards/slack_card.py

def build_slack_blocks(
    *, flow_title, node_title, applicant_name, actor_name,
    deadline_at, description, deeplinks,
) -> list[dict]:
    """构造 Slack Block Kit blocks"""
    elements = [
        {
            "type": "button",
            "text": {"type": "plain_text", "text": _slack_label_for(dl["action"])},
            "url": dl["url"],
            "style": _slack_style_for(dl["action"]),  # primary / danger / default
        }
        for dl in deeplinks
    ]
    return [
        {"type": "header", "text": {"type": "plain_text", "text": f"审批待办：{flow_title}"}},
        {
            "type": "section",
            "fields": [
                {"type": "mrkdwn", "text": f"*节点*\n{node_title}"},
                {"type": "mrkdwn", "text": f"*申请人*\n{applicant_name}"},
                {"type": "mrkdwn", "text": f"*审批人*\n{actor_name}"},
                {"type": "mrkdwn", "text": f"*截止时间*\n{deadline_at}"},
            ],
        },
        {"type": "section", "text": {"type": "mrkdwn", "text": f"*详情*\n{description}"}},
        {"type": "divider"},
        {"type": "actions", "elements": elements},
    ]


def _slack_style_for(action: str) -> str | None:
    return {"approve": "primary", "reject": "danger"}.get(action)  # None = default
```

# backend/app/agent_builder/notification/cards/mattermost_card.py

def build_mattermost_attachment(
    *, flow_title, node_title, applicant_name, actor_name,
    deadline_at, description, deeplinks,
) -> dict:
    """构造 Mattermost attachment（含 actions 数组）"""
    actions = [
        {
            "id": dl["action"],
            "name": _mm_label_for(dl["action"]),
            "integration": {
                "url": dl["url"],
                "context": {"action": dl["action"]},
            },
            "style": _mm_style_for(dl["action"]),  # good / danger / default
        }
        for dl in deeplinks
    ]
    text = f"""### 审批待办：{flow_title}

- **节点**: {node_title}
- **申请人**: {applicant_name}
- **审批人**: {actor_name}
- **截止时间**: {deadline_at}

**详情**:
{description}
"""
    return {
        "fallback": f"审批待办：{flow_title}",
        "color": "#2196F3",
        "text": text,
        "actions": actions,
    }


def _mm_style_for(action: str) -> str:
    return {"approve": "good", "reject": "danger"}.get(action, "default")
```
