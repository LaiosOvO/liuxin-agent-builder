```python
# hitl_service.py 新增

class DelegateError(Exception):
    """委托业务异常基类"""
    def __init__(self, code: str, message: str) -> None:
        self.code = code  # 'depth_exceeded' / 'self_delegate' / 'circular' / 'cross_workspace' / 'recipient_not_found'
        super().__init__(message)

async def create_delegate_token(
    self,
    *,
    original_token: HitlToken,           # 原审批人 token（即将被消费）
    from_user: User,
    to_email: str,
    reason: str,
    node_state: NodeState,
    flow_instance: FlowInstance,
    ip: str,
    ua: str,
    timeout_seconds: int,
) -> tuple[list[HitlToken], int]:        # (new_tokens, depth)
    """委托原 token 给 to_email 用户。

    校验：
    1. to_email 用户存在 + 同 workspace（DelegateError code=recipient_not_found / cross_workspace）
    2. to_email != from_user.email（DelegateError self_delegate）
    3. to_email user_id 不在 approval_chain.approvers 内（DelegateError circular）
    4. 委托链深度 ≤ 3（DelegateError depth_exceeded）

    副作用（外层事务）：
    1. 不消费 original_token —— 调用方（API 层）已通过 store.consume(jti) 消费
       （原因：消费需 advisory_lock，API 层已持有；这里仅创建新 token + 写 records）
    2. 创建新 token 给被委托人（同 allowed_actions：approve/return/reject）
       deadline_at = now + timeout_seconds（重置）
    3. 写 node_state.payload.records 加 delegate 类型记录
    4. 更新 payload.approval_chain.delegated[from_user.id] = {to: to_email_id, depth: new_depth}
    """
    # 校验深度
    chain = (node_state.payload or {}).get("approval_chain", {})
    delegated = chain.get("delegated", {}) or {}
    prev_depth = delegated.get(str(from_user.id), {}).get("depth", 0)
    new_depth = prev_depth + 1
    if new_depth > 3:
        raise DelegateError("depth_exceeded", "委托链已超 3 层（防责任稀释）")

    # 查 recipient
    to_user_stmt = (
        select(User)
        .join(UserWorkspaceRole, UserWorkspaceRole.user_id == User.id)
        .where(
            User.email == to_email,
            UserWorkspaceRole.workspace_id == flow_instance.workspace_id,
            User.status == "active",
        )
        .limit(1)
    )
    to_user = (await self.db.execute(to_user_stmt)).scalar_one_or_none()
    if to_user is None:
        raise DelegateError("recipient_not_found", f"被委托人 {to_email} 不在当前 workspace 内")
    if to_user.id == from_user.id:
        raise DelegateError("self_delegate", "不能委托给自己")
    approvers_ids = {UUID(a) for a in chain.get("approvers", [])}
    if to_user.id in approvers_ids:
        raise DelegateError("circular", f"{to_email} 已在审批链内（防循环）")

    # 创建新 token
    expires_at = datetime.now(timezone.utc) + timedelta(seconds=timeout_seconds)
    new_tokens = [
        HitlToken(
            jti=uuid4(),
            instance_id=flow_instance.id,
            node_state_id=node_state.id,
            actor_id=to_user.id,
            action=action,
            expires_at=expires_at,
        )
        for action in ("approve", "return", "reject")
    ]
    self.db.add_all(new_tokens)
    await self.db.flush()

    # 写 records + delegated
    old_payload = node_state.payload or {}
    delegate_record = {
        "actor_id": str(from_user.id),
        "actor_email": from_user.email,
        "action": "delegate",
        "reason": reason,
        "form_data": {},
        "ts": datetime.now(timezone.utc).isoformat(),
        "ip": ip,
        "ua": ua,
        "delegate_to_id": str(to_user.id),
        "delegate_to_email": to_email,
        "depth": new_depth,
    }
    new_chain = {
        **chain,
        "delegated": {
            **delegated,
            str(from_user.id): {"to": str(to_user.id), "depth": new_depth},
        },
    }
    new_payload = {
        **old_payload,
        "records": [*(old_payload.get("records") or []), delegate_record],
        "approval_chain": new_chain,
    }
    node_state.payload = new_payload

    return new_tokens, new_depth
```

```python
# api/hitl.py 在现有 POST /hitl/action/<jwt> 路由中加入 op 分支

@router.post("/action/{jwt_token}")
async def submit_hitl_action(
    jwt_token: str,
    body: HitlActionRequest,           # 现有
    op: str = Query("submit"),         # NEW: ?op=delegate | submit
    ...
):
    # 解码 jwt → payload
    payload = decode_hitl_jwt(jwt_token)

    if op == "delegate":
        return await _handle_delegate(payload, body, db, redis, ...)

    # 现有 submit 逻辑（不动）
    ...

async def _handle_delegate(payload, body: DelegateRequest, db, redis, ...):
    # 1. advisory_xact_lock（同 submit）
    # 2. 加载 flow_instance + node_state
    # 3. consume 原 token（HitlTokenStore.consume — 设置 used_at + ip + ua）
    # 4. invalidate_siblings（同 submit — 同 node_state 其他 token 全失效）
    # 5. 调 HitlService.create_delegate_token → 返回 (new_tokens, depth)
    # 6. 发通知（沿用 node config.notify_channels — Phase 4 这里先只发邮件，多通道等 04-11 完成后回填）
    # 7. audit_log action='hitl.delegate' + decision='delegate' + meta
    # 8. commit
    # 9. structured log 'hitl.chain.delegate'

    return DelegateResponse(
        ok=True,
        new_token_count=len(new_tokens),
        depth=depth,
        recipient_email=body.to_email,
    )
```
