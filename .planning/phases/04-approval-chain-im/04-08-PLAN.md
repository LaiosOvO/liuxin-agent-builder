```python
# backend/app/agent_builder/notification/cards/dingtalk_card.py

def build_dingtalk_action_card(
    *, flow_title, node_title, applicant_name, actor_name,
    deadline_at, description, deeplinks,
) -> dict:
    """构造钉钉 ActionCard JSON

    参考钉钉文档 ActionCard 结构
    """
    markdown = f"""### 审批待办：{flow_title}

**节点**：{node_title}
**申请人**：{applicant_name}
**审批人**：{actor_name}
**截止时间**：{deadline_at}

**详情**：
{description}
"""
    btns = [
        {
            "title": _dingtalk_label_for(dl["action"]),
            "actionURL": dl["url"],
        }
        for dl in deeplinks
    ]
    return {
        "msgtype": "actionCard",
        "actionCard": {
            "title": f"审批待办：{flow_title}",
            "markdown": markdown,
            "btnOrientation": "0",  # 横排
            "btns": btns,
        },
    }


def _dingtalk_label_for(action: str) -> str:
    return {"approve": "同意", "return": "退回", "reject": "拒绝", "submit": "提交"}.get(action, action)
```
